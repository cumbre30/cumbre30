    79→/** Rumbo (0-360) entre dos puntos. */
    80→function bearingOf(a: LatLng, b: LatLng): number {
    81→  const toR = Math.PI / 180;
    82→  const y = Math.sin((b.lon - a.lon) * toR) * Math.cos(b.lat * toR);
    83→  const x = Math.cos(a.lat * toR) * Math.sin(b.lat * toR) - Math.sin(a.lat * toR) * Math.cos(b.lat * toR) * Math.cos((b.lon - a.lon) * toR);
    84→  return (Math.atan2(y, x) / toR + 360) % 360;
    85→}

    87→/** Desplaza un punto `distM` metros según un rumbo (modo libre). */
    88→function offsetAlong(p: LatLng, headingDeg: number, distM: number): LatLng {
    89→  const r = (headingDeg * Math.PI) / 180;
    90→  const mPerDegLat = 111_320;
    91→  const mPerDegLon = 111_320 * Math.cos((p.lat * Math.PI) / 180);
    92→  return {
    93→    lat: p.lat + (Math.cos(r) * distM) / mPerDegLat,
    94→    lon: p.lon + (Math.sin(r) * distM) / mPerDegLon,
    95→  };
    96→}

   420→

   422→  useEffect(() => {
   423→    const map = mapRef.current;
   424→    if (!map || !ready) return;
   425→    if (props.dest) {
   426→      if (!flagRef.current) {
   427→        flagRef.current = new maplibregl.Marker({ element: buildFlagEl() }).setLngLat([props.dest.lon, props.dest.lat]).addTo(map);
   428→      } else {
   429→        flagRef.current.setLngLat([props.dest.lon, props.dest.lat]);
   430→      }
   431→    } else if (flagRef.current) {
   432→      flagRef.current.remove();
   433→      flagRef.current = null;
   434→    }
   435→
   436→    const badge = props.nextStep && props.nextStep.icon !== "end" ? props.nextStep : null;
   437→    if (badge) {
   438→      if (!badgeRef.current) {
   439→        badgeRef.current = new maplibregl.Marker({ element: buildBadgeEl(badge.icon) }).setLngLat([badge.point.lon, badge.point.lat]).addTo(map);
   440→        badgeIconRef.current = badge.icon;
   441→      } else {
   442→        badgeRef.current.setLngLat([badge.point.lon, badge.point.lat]);
   443→        if (badgeIconRef.current !== badge.icon) {
   444→          // reemplazo del elemento: el DOM interno cambia de glifo
   445→          const oldEl = badgeRef.current.getElement();
   446→          const newEl = buildBadgeEl(badge.icon);
   447→          if (oldEl.parentNode) oldEl.parentNode.replaceChild(newEl, oldEl);
   448→          badgeIconRef.current = badge.icon;
   449→        }
   450→      }
   451→    } else if (badgeRef.current) {
   452→      badgeRef.current.remove();
   453→      badgeRef.current = null;
   454→      badgeIconRef.current = null;
   455→    }
   456→  }, [ready, props.dest, props.nextStep]);
   457→
   458→  // incidencias de la comunidad
   459→  useEffect(() => {
   460→    const map = mapRef.current;
   461→    if (!map || !ready) return;
   462→    for (const m of incidentMarkersRef.current) m.remove();
   463→    incidentMarkersRef.current = props.incidents.map((inc) =>
   464→      new maplibregl.Marker({ element: buildIncidentEl(inc.type) }).setLngLat([inc.lon, inc.lat]).addTo(map)
   465→    );
   466→  }, [ready, props.incidents]);
   467→
   468→  // ---- datos GeoJSON ----
   469→  useEffect(() => {
   470→    const map = mapRef.current;
   471→    if (!map || !ready) return;
   472→
   473→    // recorrido del ride
   474→    const trackSrc = map.getSource("track") as maplibregl.GeoJSONSource | undefined;
   475→    trackSrc?.setData(props.track.length > 1 ? toFc([lineFeat(props.track)]) : toFc([]));
   476→
   477→    if (!props.onRoute || !props.route) {
   478→      (map.getSource("route-remaining") as maplibregl.GeoJSONSource | undefined)?.setData(toFc([]));
   479→      (map.getSource("route-traveled") as maplibregl.GeoJSONSource | undefined)?.setData(toFc([]));
   480→      return;
   481→    }
   482→
   483→    const r = props.route;
   484→    const from = props.nearestIdx ?? 0;
   485→    const remainingGroups = groupSegments(r.coords, r.flags, from, r.coords.length - 2);
   486→    const traveledGroups = props.nearestIdx !== null ? groupSegments(r.coords, r.flags, 0, from - 1) : [];
   487→
   488→    const remainingFeats: GeoFeature[] = [
   489→      // casing bajo la ruta completa restante
   490→      lineFeat(r.coords.slice(from), { kind: "casing", ciclo: false }),
   491→      ...remainingGroups.map((g) => lineFeat(g.coords, { kind: "seg", ciclo: g.ciclovia })),
   492→    ];
   493→    const traveledFeats = traveledGroups.map((g) => lineFeat(g.coords, { ciclo: g.ciclovia }));
   494→
   495→    (map.getSource("route-remaining") as maplibregl.GeoJSONSource | undefined)?.setData(toFc(remainingFeats));
   496→    (map.getSource("route-traveled") as maplibregl.GeoJSONSource | undefined)?.setData(toFc(traveledFeats));
   497→  }, [ready, props.onRoute, props.route, props.nearestIdx, props.track]);
   498→
   499→  // ---- cámara de navegación ----
   500→  useEffect(() => {
   501→    const map = mapRef.current;
   502→    if (!map || !ready) return;
   503→
   504→    // Sin posición: encuadra la ruta completa (vista previa plana)
   505→    if (!props.position) {
   506→      if (props.onRoute && props.route) {
   507→        const lons = props.route.coords.map((c) => c.lon);
   508→        const lats = props.route.coords.map((c) => c.lat);
   509→        map.easeTo({
   510→          center: [(Math.min(...lons) + Math.max(...lons)) / 2, (Math.min(...lats) + Math.max(...lats)) / 2],
   511→          zoom: 13.8,
   512→          pitch: 0,
   513→          bearing: 0,
   514→          duration: 700,
   515→        });
   516→      }
   517→      return;
   518→    }
   519→
   520→    const h = boxRef.current?.clientHeight ?? 260;
   521→    const onRouteClean = props.onRoute && !props.offRoute && props.remainingM >= ARRIVAL_M && props.route && props.nearestIdx !== null;
   522→    let target: LatLng;
   523→    if (onRouteClean) {
   524→      target = pointAhead(props.route!.coords, props.nearestIdx!, LOOKAHEAD_NAV_M);
   525→    } else if (props.heading !== null) {
   526→      target = offsetAlong(props.position, props.heading, props.onRoute ? LOOKAHEAD_NAV_M : LOOKAHEAD_FREE_M);
   527→    } else if (props.route && props.nearestIdx !== null && props.nearestIdx < props.route.coords.length - 1) {
   528→      const br = bearingOf(props.route.coords[props.nearestIdx], props.route.coords[props.nearestIdx + 1]);
   529→      target = offsetAlong(props.position, br, LOOKAHEAD_NAV_M);
   530→    } else {
   531→      target = props.position;
   532→    }
   533→
   534→    const wantedBearing = props.heading !== null ? props.heading : map.getBearing();
   535→    const cur = map.getBearing();
   536→    let b = wantedBearing;
   537→    // giro por el camino corto (evita vueltas de 340°)
   538→    while (b - cur > 180) b -= 360;
   539→    while (b - cur < -180) b += 360;
   540→
   541→    const camZoom = props.onRoute && props.remainingM < ARRIVAL_M ? ARRIVAL_ZOOM : NAV_ZOOM;
   542→
   543→    if (!firstFixRef.current) {
   544→      firstFixRef.current = true;
   545→      map.jumpTo({
   546→        center: [target.lon, target.lat],
   547→        bearing: b,
   548→        pitch: NAV_PITCH,
   549→        zoom: camZoom,
   550→        padding: { top: 0, bottom: Math.round(h * 0.3), left: 0, right: 0 },
   551→      });
   552→      return;
   553→    }
   554→
   555→    map.easeTo({
   556→      center: [target.lon, target.lat],
   557→      bearing: b,
   558→      pitch: NAV_PITCH,
   559→      zoom: camZoom,
   560→      padding: { top: 0, bottom: Math.round(h * 0.3), left: 0, right: 0 },
   561→      duration: 680,
   562→      easing: (t) => t,
   563→    });
   564→  }, [ready, props.position, props.heading, props.onRoute, props.offRoute, props.remainingM, props.nearestIdx, props.route]);
   565→
   566→  // rotación del pin flecha respecto al mapa (queda alineado al rumbo)
   567→  useEffect(() => {
   568→    const map = mapRef.current;
   569→    if (!map || !ready) return;
   570→    const apply = () => {
   571→      const m = puckRef.current;
   572→      const arrow = arrowRef.current;
   573→      if (!m || !arrow) return;
   574→      const hdg = props.heading ?? 0;
   575→      arrow.style.transform = `rotate(${hdg - map.getBearing()}deg)`;
   576→    };
   577→    apply();
   578→    map.on("rotate", apply);
   579→    return () => {
   580→      map.off("rotate", apply);
   581→    };
   582→  }, [ready, props.heading, props.position]);
   583→
   584→  // vista a un punto (localización sin ride)
   585→  useEffect(() => {
   586→    const map = mapRef.current;
   587→    if (!map || !ready || !props.locateTarget) return;
   588→    map.flyTo({ center: [props.locateTarget.lon, props.locateTarget.lat], zoom: Math.max(map.getZoom(), 16), duration: 800 });
   589→  }, [ready, props.locateTarget]);
   590→
   591→  return (
   592→    // position absolute por estilo inline: el CSS de maplibre usa
   593→    // .maplibregl-map{position:relative} y pisa la utilidad `absolute`
   594→    <div ref={boxRef} style={{ position: "absolute", inset: 0 }} />
   595→  );
   596→}

   598→// ============================================================
   599→// Componente principal: lienzo + capa Waze (banner, barra, etc.)
   600→// ============================================================
   601→export default function MiniMap({ position, heading, track, accuracyM, userPin, nav, speedKmh = 0 }: Props) {
   602→  const incidents = useIncidentFeed();
   603→  const [pickMode, setPickMode] = useState(false);
   604→  const [locateTarget, setLocateTarget] = useState<LatLng | null>(null);
   605→  const [reportOpen, setReportOpen] = useState(false);
   606→  const [status, setStatus] = useState<string | null>(null);
   607→  const route = nav.route;
   608→  const dest = nav.dest;
   609→
   610→  const webglOk = useMemo(() => {
   611→    try {
   612→      return webglSupported();
   613→    } catch {
   614→      return false;
   615→    }
   616→  }, []);
   617→
   618→  const pick = (p: LatLng) => {
   619→    nav.setDestination({ name: "Punto en el mapa", position: p });
   620→    setPickMode(false);
   621→  };
   622→
   623→  const locate = () => {
   624→    if (position) {
   625→      setLocateTarget(position);
   626→      return;
   627→    }
   628→    if ("geolocation" in navigator) {
   629→      navigator.geolocation.getCurrentPosition(
   630→        (pos) => setLocateTarget({ lat: pos.coords.latitude, lon: pos.coords.longitude }),
   631→        () => {},
   632→        { enableHighAccuracy: true, timeout: 8000 }
   633→      );
   634→    }
   635→  };
   636→
   637→  const onRoute = !!route && route.coords.length > 1;
   638→
   639→  const progress = useMemo(
   640→    () => (onRoute && position ? routeProgress(route!, position) : null),
   641→    [onRoute, route, position]
   642→  );
   643→
   644→  const nextStep = progress?.nextStep ?? route?.steps[0] ?? null;
   645→  const laterStep = onRoute && progress && route ? route.steps[progress.passedCount + 1] ?? null : null;
   646→  const offRoute = !!progress && progress.offRouteM > 140;
   647→  const remainingM = progress?.remainingM ?? route?.distanceM ?? 0;
   648→
   649→  const currentStreet = onRoute && progress ? streetAt(route!, progress.nearestIdx) : "";
   650→
   651→  const spd = speedKmh > 4 ? speedKmh : 15;
   652→  const etaMin = remainingM > 0 ? Math.max(1, Math.round((remainingM / 1000 / spd) * 60)) : 0;
   653→  const arrivalClock = useMemo(
   654→    () => new Date(Date.now() + etaMin * 60000).toLocaleTimeString("es-CL", { hour: "2-digit", minute: "2-digit", hourCycle: "h23" }),
   655→    [etaMin]
   656→  );
   657→
   658→  // ---- Reporte rápido de incidencias (botón amarillo) ----
   659→  const reportIncident = async (type: IncidentTypeId) => {
   660→    if (!position) {
   661→      setStatus("Aún sin GPS: no podemos fijar la ubicación del reporte.");
   662→      setReportOpen(false);
   663→      setTimeout(() => setStatus(null), 4000);
   664→      return;
   665→    }
   666→    setReportOpen(false);
   667→    const payload = { type, lat: position.lat, lon: position.lon };
   668→    try {
   669→      const r = await fetch("/api/incidents", {
   670→        method: "POST",
   671→        headers: { "Content-Type": "application/json" },
   672→        body: JSON.stringify(payload),
   673→      });
   674→      const d = (await r.json()) as { incident?: IncidentItem; error?: string };
   675→      if (!r.ok || !d.incident) throw new Error(d.error ?? "error");
   676→      addIncident(d.incident);
   677→      setStatus("Incidencia reportada ✓ visible para toda la comunidad");
   678→    } catch {
   679→      const local: IncidentItem = {
   680→        id: crypto.randomUUID(),
   681→        ...payload,
   682→        note: null,
   683→        createdAt: new Date().toISOString(),
   684→        username: "tú (local)",
   685→        authorPin: userPin ?? DEFAULT_PIN,
   686→        mine: true,
   687→        localOnly: true,
   688→      };
   689→      addIncident(local);
   690→      saveLocalIncidents([local, ...loadLocalIncidents()]);
   691→      setStatus("Sin conexión: reporte guardado en este dispositivo.");
   692→    }
   693→    setTimeout(() => setStatus(null), 5000);
   694→  };
   695→
   696→  return (
   697→    <div className={`relative rounded-xl overflow-hidden border border-zinc-800 z-0 c30-minimap ${onRoute ? "h-80" : "h-52 sm:h-56"}`}>
   698→      {webglOk ? (
   699→        <NavCanvas
   700→          position={position}
   701→          heading={heading}
   702→          track={track}
   703→          onRoute={onRoute}
   704→          route={onRoute ? route : null}
   705→          nearestIdx={progress ? progress.nearestIdx : null}
   706→          offRoute={offRoute}
   707→          remainingM={remainingM}
   708→          nextStep={nextStep}
   709→          dest={dest ? dest.position : null}
   710→          pickMode={pickMode}
   711→          onPick={pick}
   712→          userPin={userPin}
   713→          incidents={incidents}
   714→          locateTarget={locateTarget}
   715→        />
   716→      ) : (
   717→        <div className="absolute inset-0 flex flex-col items-center justify-center gap-1 bg-zinc-950 text-center px-6">
   718→          <RouteIcon className="h-6 w-6 text-zinc-600" />
   719→          <p className="text-sm text-zinc-400 font-medium">La vista 3D necesita WebGL</p>
   720→          <p className="text-[11px] text-zinc-600">Actualiza Chrome/Sistema Web Android para ver la navegación.</p>
   721→        </div>
   722→      )}
   723→
   724→      {/* ===== Banner de maniobra (modo ruta), estilo Waze ===== */}
   725→      {onRoute && nextStep && (
   726→        <div className="absolute top-0 inset-x-0 z-[500] bg-zinc-950/95 border-b border-zinc-800 px-3 pt-1.5 pb-2 flex items-center gap-3 pointer-events-none">
   727→          <TurnArrow icon={nextStep.icon} className="h-14 w-14 shrink-0 text-zinc-50" />
   728→          <div className="min-w-0 flex-1">
   729→            <div className="font-mono text-[26px] font-bold text-zinc-50 leading-none tabular-nums">
   730→              {nextStep.icon === "end" ? "Llegaste" : fmtKmM(progress?.distToNextStepM ?? nextStep.distM)}
   731→            </div>
   732→            <div className="text-lg font-bold text-sky-400 leading-tight truncate mt-0.5">
   733→              {nextStep.icon === "end" ? dest?.name ?? "Destino" : streetOfStep(nextStep.text)}
   734→            </div>
   735→            {offRoute && <div className="text-[11px] text-amber-300 font-semibold mt-0.5">desviado, recalculando…</div>}
   736→          </div>
   737→          {laterStep && laterStep.icon !== "end" && <TurnArrow icon={laterStep.icon} className="h-7 w-7 shrink-0 mt-1 text-zinc-500" />}
   738→        </div>
   739→      )}
   740→
   741→      {/* Estado de cálculo de ruta */}
   742→      {nav.loading && (
   743→        <div className="absolute top-2 left-2 z-[520] rounded-full bg-zinc-950/90 border border-sky-700 px-2.5 py-1 text-[10px] font-semibold text-sky-300 flex items-center gap-1.5 pointer-events-none">
   744→          <span className="inline-block h-2 w-2 rounded-full bg-sky-400 animate-pulse" />
   745→          Calculando ruta…
   746→        </div>
   747→      )}
   748→
   749→      {/* Leyenda compacta de la ruta (solo modo ruta) */}
   750→      {onRoute && !nav.loading && (
   751→        <div className="absolute left-2 top-[92px] z-[500] rounded-md bg-zinc-950/90 border border-zinc-800 px-2 py-1 flex items-center gap-2 pointer-events-none">
   752→          <span className="inline-block h-[3px] w-4 rounded-full shrink-0" style={{ background: COLOR_CICLO }} />
   753→          <span className="text-[10px] text-zinc-300">Ciclovía {!!route?.cicloviaPct ? `${route.cicloviaPct}%` : ""}</span>
   754→          <span className="text-[10px] text-zinc-600">·</span>
   755→          <span className="inline-block h-[3px] w-4 rounded-full shrink-0" style={{ background: COLOR_VIA }} />
   756→          <span className="text-[10px] text-zinc-300">Compartida</span>
   757→        </div>
   758→      )}
   759→
   760→      {/* Píldora con la calle actual sobre el pin (modo ruta) */}
   761→      {onRoute && currentStreet && !offRoute && !(progress && progress.remainingM < ARRIVAL_M) && (
   762→        <div className="absolute bottom-[108px] inset-x-0 z-[500] flex justify-center pointer-events-none">
   763→          <span className="rounded-full bg-zinc-100 text-zinc-900 text-[11px] font-semibold px-3 py-1 shadow-lg max-w-[70%] truncate">
   764→            {currentStreet}
   765→          </span>
   766→        </div>
   767→      )}
   768→
   769→      {/* Controles flotantes */}
   770→      <div className="absolute top-2 right-2 z-[500] flex flex-col gap-1.5">
   771→        <button
   772→          onClick={locate}
   773→          aria-label="Localizarme"
   774→          title="Centrar en mi ubicación"
   775→          className="h-8 w-8 rounded-md bg-zinc-950/90 border border-zinc-700 text-zinc-100 flex items-center justify-center hover:border-emerald-500"
   776→        >
   777→          <Crosshair className="h-4 w-4" />
   778→        </button>
   779→        <button
   780→          onClick={() => setPickMode((v) => !v)}
   781→          aria-label="Elegir destino en el minimapa"
   782→          title="Toca el minimapa para fijar destino"
   783→          className={`h-8 w-8 rounded-md border flex items-center justify-center ${
   784→            pickMode ? "bg-sky-600 border-sky-300 text-white" : "bg-zinc-950/90 border-zinc-700 text-zinc-100 hover:border-sky-500"
   785→          }`}
   786→        >
   787→          <Flag className="h-4 w-4" />
   788→        </button>
   789→        {dest && (
   790→          <button
   791→            onClick={() => nav.setDestination(null)}
   792→            aria-label="Quitar destino"
   793→            title="Quitar la ruta"
   794→            className="h-8 w-8 rounded-md bg-zinc-950/90 border border-zinc-700 text-zinc-100 flex items-center justify-center hover:border-red-500"
   795→          >
   796→            <X className="h-4 w-4" />
   797→          </button>
   798→        )}

   800→
   801→      {/* ===== Barra inferior tipo Waze — SIN velocímetro ===== */}
   802→      <div className="absolute bottom-0 inset-x-0 z-[500] bg-zinc-950/95 border-t border-zinc-800 px-2.5 py-2 flex items-center gap-2.5">
   803→        {onRoute ? (
   804→          <div className="min-w-0 flex-1 text-center">
   805→            <div className="font-mono text-xl font-bold text-zinc-50 leading-none tabular-nums">{etaMin > 0 ? arrivalClock : "—"}</div>
   806→            <div className="text-[11px] text-zinc-400 leading-tight mt-0.5 truncate">
   807→              {etaMin > 0 ? `${etaMin} min · ${fmtKmM(remainingM)}` : "Ruta lista"}
   808→            </div>
   809→          </div>
   810→        ) : (
   811→          <div className="min-w-0 flex-1 flex items-center gap-1.5 rounded-lg border border-zinc-800 bg-zinc-900 px-2 py-1.5">
   812→            <RouteIcon className="h-3.5 w-3.5 text-sky-400 shrink-0" />

   816→              onChange={(e) => {

   819→                else nav.setDestination(null);
   820→              }}
   821→              className="bg-transparent text-[11px] text-zinc-200 outline-none w-full min-w-0"
   822→            >
   823→              <option value="" className="bg-zinc-900">Ir a…</option>
   824→              {NAV_DESTINATIONS.map((d) => (
   825→                <option key={d.id} value={d.id} className="bg-zinc-900">{d.name}</option>
   826→              ))}
   827→              {dest && <option value="mapa" className="bg-zinc-900">Punto en el mapa</option>}
   828→            </select>
   829→          </div>
   830→        )}
   831→
   832→        {/* botón amarillo de reporte de incidencias */}
   833→        <button
   834→          onClick={() => setReportOpen((v) => !v)}
   835→          aria-label="Reportar incidencia en mi posición"
   836→          title="Reportar incidencia en mi posición"
   837→          className="relative h-11 w-11 rounded-xl bg-amber-400 hover:bg-amber-300 active:scale-95 transition text-zinc-950 flex items-center justify-center shrink-0"
   838→        >
   839→          <TriangleAlert className="h-6 w-6" strokeWidth={2.4} />
   840→          <Plus className="absolute bottom-0 right-0.5 h-3.5 w-3.5" strokeWidth={3.5} />
   841→        </button>
   842→      </div>
   843→
   844→      {/* Hoja de reporte rápido */}
   845→      {reportOpen && (
   846→        <div className="absolute bottom-[60px] inset-x-2 z-[650] rounded-xl bg-zinc-950/95 border border-zinc-700 shadow-2xl p-3">
   847→          <div className="flex items-center justify-between mb-2">
   848→            <span className="text-xs font-bold text-zinc-100">Reportar incidencia {position ? "en tu posición" : "(sin GPS aún)"}</span>
   849→            <button onClick={() => setReportOpen(false)} aria-label="Cerrar reporte" className="text-zinc-400 hover:text-zinc-100">
   850→              <X className="h-4 w-4" />
   851→            </button>
   852→          </div>
   853→          <div className="grid grid-cols-2 gap-1.5">
   854→            {INCIDENT_TYPES.map((t) => (
   855→              <button
   856→                key={t.id}
   857→                onClick={() => reportIncident(t.id)}
   858→                className="flex items-center gap-1.5 rounded-lg border border-zinc-800 bg-zinc-900 px-2 py-1.5 text-left text-[11px] text-zinc-200 hover:border-zinc-500"
   859→              >
   860→                <span aria-hidden>{t.emoji}</span>
   861→                <span className="truncate">{t.label}</span>
   862→              </button>
