   231→   231→function NavCanvas(props: NavCanvasProps) {
   232→   232→  const boxRef = useRef<HTMLDivElement>(null);
   233→   233→  const mapRef = useRef<MlMap | null>(null);
   234→   234→  const [ready, setReady] = useState(false);

   240→   240→  useEffect(() => {
   241→   241→    pickModeRef.current = props.pickMode;
   242→   242→    onPickRef.current = props.onPick;
   243→   243→  }, [props.pickMode, props.onPick]);

   245→   245→  // ---- init ----

   247→   247→    if (!boxRef.current || mapRef.current) return;

   259→   259→      scrollZoom: false,
   260→   260→      doubleClickZoom: false,
   261→   261→      touchZoomRotate: false,

   915→   915→          <NativeMapCanvas

   917→   917→            mode="nav"
   918→   918→            position={position}
   919→   919→            heading={heading}
   920→   920→            track={track}

   922→   922→            onRoute={onRoute}
   923→   923→            route={onRoute ? route : null}
   924→   924→            nearestIdx={progress ? progress.nearestIdx : null}
   925→   925→            remainingM={remainingM}
   926→   926→            nextStep={nextStep}
