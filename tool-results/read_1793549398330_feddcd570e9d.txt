   124→   124→/** Centra y sigue la posición en modo libre (norte arriba, zoom ciclócomputador). */
   125→   125→function FollowFree({ position }: { position: LatLng | null }) {
   126→   126→  const map = useMap();
   127→   127→  const init = useRef(false);
   128→   128→  useEffect(() => {
   129→   129→    if (!position) return;
   130→   130→    if (map.getBearing && map.getBearing() !== 0) map.setBearing(0);
   131→   131→    if (!init.current) {
   132→   132→      init.current = true;
   133→   133→      map.setView([position.lat, position.lon], 17, { animate: false });
   134→   134→      return;
   135→   135→    }
   136→   136→    map.setView([position.lat, position.lon], map.getZoom(), { animate: true, duration: 0.6 });
   137→   137→  }, [position, map]);
   138→   138→  return null;
   139→   139→}

   142→   142→function FitRoute({ coords }: { coords: LatLng[] | null }) {
   143→   143→  const map = useMap();
   144→   144→  useEffect(() => {
   145→   145→    if (!coords || coords.length < 2) return;
   146→   146→    if (map.getBearing && map.getBearing() !== 0) map.setBearing(0);
   147→   147→    const bounds = L.latLngBounds(coords.map((c) => [c.lat, c.lon] as [number, number]));
   148→   148→    map.fitBounds(bounds, { paddingTopLeft: [16, 92], paddingBottomRight: [16, 84] });
   149→   149→  }, [coords, map]);
   150→   150→  return null;
   151→   151→}

   158→   158→function NavCamera({ position, bearing, target, zoom }: { position: LatLng | null; bearing: number; target: LatLng; zoom: number }) {
   159→   159→  const map = useMap();
   160→   160→  const init = useRef(false);
   161→   161→  useEffect(() => {
   162→   162→    if (!position) return;
   163→   163→    const b = wrap360(bearing);
   164→   164→    if (!init.current) {
   165→   165→      init.current = true;
   166→   166→      map.setBearing(b);
   167→   167→      map.setView([target.lat, target.lon], zoom, { animate: false });
   168→   168→      return;
   169→   169→    }
   170→   170→    map.setBearing(b, { animate: true, duration: 0.4 });
   171→   171→    map.setView([target.lat, target.lon], zoom, { animate: true, duration: 0.6 });
   172→   172→  }, [position, map]); // target/zoom/derivados del mismo tick de posición
   173→   173→  return null;
   174→   174→}

   198→   198→export default function MiniMap({ position, heading, track, accuracyM, userPin, nav, speedKmh = 0 }: Props) {
   199→   199→  const incidents = useIncidentFeed();
   200→   200→  const [pickMode, setPickMode] = useState(false);

   202→   202→  const [reportOpen, setReportOpen] = useState(false);
   203→   203→  const [status, setStatus] = useState<string | null>(null);
   204→   204→  const route = nav.route;
   205→   205→  const dest = nav.dest;
   206→   206→
   207→   207→  const pick = (p: LatLng) => {
   208→   208→    nav.setDestination({ name: "Punto en el mapa", position: p });
   209→   209→    setPickMode(false);
   210→   210→  };
   211→   211→
   212→   212→  const locate = () => {
   213→   213→    if (position) {
   214→   214→      setLocateTarget(position);
   215→   215→      return;
   216→   216→    }
   217→   217→    if ("geolocation" in navigator) {
   218→   218→      navigator.geolocation.getCurrentPosition(
   219→   219→        (pos) => setLocateTarget({ lat: pos.coords.latitude, lon: pos.coords.longitude }),
   220→   220→        () => {},
   221→   221→        { enableHighAccuracy: true, timeout: 8000 }
   222→   222→      );
   223→   223→    }
   224→   224→  };
   225→   225→
   226→   226→  const onRoute = !!route && route.coords.length > 1;
   227→   227→
   228→   228→  // Progreso vivo sobre la ruta (restante + desvío)
   229→   229→  const progress = useMemo(
   230→   230→    () => (onRoute && position ? routeProgress(route!, position) : null),
   231→   231→    [onRoute, route, position]
   232→   232→  );
   233→   233→
   234→   234→  const nextStep = progress?.nextStep ?? route?.steps[0] ?? null;
   235→   235→  const laterStep = onRoute && progress && route ? route.steps[progress.passedCount + 1] ?? null : null;
   236→   236→  const offRoute = !!progress && progress.offRouteM > 140;
   237→   237→  const remainingM = progress?.remainingM ?? route?.distanceM ?? 0;
   238→   238→
   239→   239→  // Ruta COMPLETA (inicio → fin) partida por condición: recorrida (tenue) + restante (plena)
   240→   240→  const traveledGroups = useMemo(
   241→   241→    () => (onRoute && progress ? groupSegments(route!.coords, route!.flags, 0, progress.nearestIdx - 1) : []),
   242→   242→    [onRoute, progress, route]
   243→   243→  );
   244→   244→  const remainingGroups = useMemo(
   245→   245→    () => (onRoute ? groupSegments(route!.coords, route!.flags, progress ? progress.nearestIdx : 0, route!.coords.length - 2) : []),
   246→   246→    [onRoute, progress, route]
   247→   247→  );
   248→   248→  const remainingLine = useMemo(
   249→   249→    () => (onRoute ? route!.coords.slice(progress ? progress.nearestIdx : 0).map((c) => [c.lat, c.lon] as [number, number]) : []),
   250→   250→    [onRoute, progress, route]
   251→   251→  );
   252→   252→
   253→   253→  // Calle actual (píldora sobre el pin)
   254→   254→  const currentStreet = onRoute && progress ? streetAt(route!, progress.nearestIdx) : "";
   255→   255→
   256→   256→  // ETA vivo: con la velocidad actual (toma 15 km/h como referencia base)
   257→   257→  const spd = speedKmh > 4 ? speedKmh : 15;
   258→   258→  const etaMin = remainingM > 0 ? Math.max(1, Math.round((remainingM / 1000) / spd * 60)) : 0;
   259→   259→  const arrivalClock = useMemo(
   260→   260→    () => new Date(Date.now() + etaMin * 60000).toLocaleTimeString("es-CL", { hour: "2-digit", minute: "2-digit", hourCycle: "h23" }),
   261→   261→    [etaMin]
   262→   262→  );
   263→   263→
   264→   264→  // ---- Reporte rápido de incidencias (botón amarillo) ----
   265→   265→  const reportIncident = async (type: IncidentTypeId) => {
   266→   266→    if (!position) {
   267→   267→      setStatus("Aún sin GPS: no podemos fijar la ubicación del reporte.");
   268→   268→      setReportOpen(false);
   269→   269→      setTimeout(() => setStatus(null), 4000);
   270→   270→      return;
   271→   271→    }
   272→   272→    setReportOpen(false);
   273→   273→    const payload = { type, lat: position.lat, lon: position.lon };
   274→   274→    try {
   275→   275→      const r = await fetch("/api/incidents", {
   276→   276→        method: "POST",
   277→   277→        headers: { "Content-Type": "application/json" },
   278→   278→        body: JSON.stringify(payload),
   279→   279→      });
   280→   280→      const d = (await r.json()) as { incident?: IncidentItem; error?: string };
   281→   281→      if (!r.ok || !d.incident) throw new Error(d.error ?? "error");
   282→   282→      addIncident(d.incident);
   283→   283→      setStatus("Incidencia reportada ✓ visible para toda la comunidad");
   284→   284→    } catch {
   285→   285→      const local: IncidentItem = {
   286→   286→        id: crypto.randomUUID(),
   287→   287→        ...payload,
   288→   288→        note: null,
   289→   289→        createdAt: new Date().toISOString(),
   290→   290→        username: "tú (local)",
   291→   291→        authorPin: userPin ?? DEFAULT_PIN,
   292→   292→        mine: true,
   293→   293→        localOnly: true,
   294→   294→      };
   295→   295→      addIncident(local);
   296→   296→      saveLocalIncidents([local, ...loadLocalIncidents()]);
   297→   297→      setStatus("Sin conexión: reporte guardado en este dispositivo.");
   298→   298→    }
   299→   299→    setTimeout(() => setStatus(null), 5000);
   300→   300→  };
   301→   301→
   302→   302→  // ----- Cámara tipo navegación -----
   303→   303→  const navBearing = onRoute && heading !== null ? wrap360(-heading) : 0;
   304→   304→  const camTarget: LatLng = useMemo(() => {
   305→   305→    if (!onRoute || !position) return position ?? PARK_CENTER_OBJ;
   306→   306→    if (offRoute || progress!.remainingM < ARRIVAL_M) return position;
   307→   307→    return pointAhead(route!.coords, progress!.nearestIdx, LOOKAHEAD_M);
   308→   308→  }, [onRoute, position, offRoute, progress, route]);
   309→   309→  const camZoom = onRoute && progress && progress.remainingM < ARRIVAL_M ? ARRIVAL_ZOOM : NAV_ZOOM;
   310→   310→
   311→   311→  return (
   312→   312→    <div className={`relative rounded-xl overflow-hidden border border-zinc-800 z-0 c30-minimap ${onRoute ? "h-72" : "h-52 sm:h-56"}`}>
   313→   313→      <MapContainer
   314→   314→        center={position ? [position.lat, position.lon] : PARK_CENTER}
   315→   315→        zoom={position ? 17 : 13}
   316→   316→        zoomControl={false}
   317→   317→        attributionControl={false}
   318→   318→        rotate
   319→   319→        bearing={0}
   320→   320→        rotateControl={false}
   321→   321→        touchRotate={false}
   322→   322→        dragging={false}
   323→   323→        scrollWheelZoom={false}
   324→   324→        doubleClickZoom={false}
   325→   325→        touchZoom={false}
   326→   326→        boxZoom={false}
   327→   327→        keyboard={false}
   328→   328→        className={`map-dark h-full w-full bg-zinc-950 ${pickMode ? "cursor-crosshair" : ""}`}
   329→   329→      >
   330→   330→        <TileLayer url="https://tile.openstreetmap.org/{z}/{x}/{y}.png" maxZoom={19} />
   331→   331→
   332→   332→        {!onRoute && (
   333→   333→          <>
   334→   334→            <Polygon
   335→   335→              positions={PARK_POLYGON.map((p) => [p.lat, p.lon])}
   336→   336→              pathOptions={{ color: "#34d399", weight: 1.5, fillColor: "#10b981", fillOpacity: 0.07 }}
   337→   337→              interactive={false}
   338→   338→            />
   339→   339→            {RISK_ZONES.map((z) => (
   340→   340→              <Circle
   341→   341→                key={z.id}
   342→   342→                center={[z.position.lat, z.position.lon]}
   343→   343→                radius={z.radiusM}
   344→   344→                pathOptions={{ color: "#f59e0b", weight: 1, fillColor: "#f59e0b", fillOpacity: 0.13 }}
   345→   345→                interactive={false}

   358→   358→              <Polyline positions={track.map((p) => [p.lat, p.lon])} pathOptions={{ color: "#a3e635", weight: 3.5, opacity: 0.9 }} interactive={false} />
   359→   359→            )}
   360→   360→          </>
   361→   361→        )}
   362→   362→
   363→   363→        {/* Incidencias de la comunidad: visibles en ambos modos */}
   364→   364→        {incidents.map((inc) => (
   365→   365→          <Marker key={inc.id} position={[inc.lat, inc.lon]} icon={miniIncidentIcon(inc.type)} interactive={false} />

   367→   367→
   368→   368→        {onRoute && (

   371→   371→            {remainingLine.length > 1 && (
   372→   372→              <Polyline positions={remainingLine} pathOptions={{ color: COLOR_CASING, weight: 10, opacity: 0.85 }} interactive={false} />

   382→   382→            ))}
   383→   383→            {/* tramo restante: violeta = ciclovía · blanco = vía compartida */}

   389→   389→                interactive={false}
   390→   390→              />

   392→   392→            {nextStep && nextStep.icon !== "end" && (
   393→   393→              <Marker position={[nextStep.point.lat, nextStep.point.lon]} icon={stepBadgeIcon(nextStep.icon)} interactive={false} />

   399→   399→        {/* Posición actual: pin elegido por el usuario o flecha por defecto */}
   400→   400→        {position && (
   401→   401→          <>

   541→   541→

   599→   599→/** Vuela a un punto una sola vez (localización sin ride activo). */
   600→   600→function FollowOnce({ target }: { target: LatLng }) {
   601→   601→  const map = useMap();
   602→   602→  useEffect(() => {
   603→   603→    map.flyTo([target.lat, target.lon], Math.max(map.getZoom(), 16), { duration: 0.8 });
   604→   604→  }, [target, map]);
   605→   605→  return null;
   606→   606→}
