   231→function NavCanvas(props: NavCanvasProps) {
   232→  const boxRef = useRef<HTMLDivElement>(null);
   233→  const mapRef = useRef<MlMap | null>(null);
   234→  const [ready, setReady] = useState(false);
   235→  const firstFixRef = useRef(false);
   236→
   237→  // refs vivas para handlers registrados una sola vez
   238→  const pickModeRef = useRef(props.pickMode);
   239→  const onPickRef = useRef(props.onPick);
   240→  useEffect(() => {
   241→    pickModeRef.current = props.pickMode;
   242→    onPickRef.current = props.onPick;
   243→  }, [props.pickMode, props.onPick]);
   244→
   245→  // ---- init ----
   246→  useEffect(() => {
   247→    if (!boxRef.current || mapRef.current) return;
   248→    const map = new maplibregl.Map({
   249→      container: boxRef.current,
   250→      style: navStyle(),
   251→      center: PARK_CENTER,
   252→      zoom: 13.5,
   253→      pitch: 0,
   254→      bearing: 0,
   255→      // Atribución compacta (obligatoria para tiles OSM/CARTO).
   256→      attributionControl: { compact: true },
   257→      dragRotate: false,
   258→      dragPan: false,
   259→      scrollZoom: false,
   260→      doubleClickZoom: false,
   261→      touchZoomRotate: false,
   262→      keyboard: false,
   263→      boxZoom: false,
   264→      maxPitch: 70,
   265→    });
   266→    mapRef.current = map;
   267→
   268→    // --- fallback de base (v1.3.2): si OSM estándar falla repetidamente
   269→    // (rate-limit 429, caída), cambiamos a CyclOSM (OSM-France) —
   270→    // prioriza funcionamiento: peor es quedarse sin mapa.
   271→    let baseTileErr = 0;
   272→    map.on("error", (e) => {
   273→      if (baseTileErr < 0) return; // ya cambiamos; no insistir
   274→      const ev = e as unknown as { sourceId?: string; error?: { message?: string } };
   275→      if (ev.sourceId && ev.sourceId !== "base") return;
   276→      const msg = ev.error?.message ?? "";
   277→      if (!ev.sourceId && !/tile|base|fetch|status|ajax/i.test(msg)) return;
   278→      baseTileErr++;
   279→      if (baseTileErr >= 5) {
   280→        const src = map.getSource("base") as maplibregl.RasterTileSource | undefined;
   281→        if (src && typeof src.setTiles === "function") {
   282→          src.setTiles(CYCLOSM_TILES);
   283→          baseTileErr = -1; // un solo cambio por sesión
   284→        }
   285→      }
   286→    });
   287→
   288→    map.on("load", () => {
   289→      map.addImage("c30-chev", makeChevronImage());
   290→
   291→      // asa de depuración/E2E (no interfiere en producción)
   292→      (window as unknown as Record<string, unknown>).__c30nav = map;
   293→
   294→      // --- fuentes ---
   295→      map.addSource("park", { type: "geojson", data: toFc([polyFeat(PARK_POLYGON.map((p) => [p.lon, p.lat] as [number, number]))]) });
   296→      map.addSource("risk", {
   297→        type: "geojson",
   298→        data: toFc(RISK_ZONES.map((z) => polyFeat(circlePolygon(z.position.lat, z.position.lon, z.radiusM), { id: z.id }))),
   299→      });
   300→      map.addSource("gates", {
   301→        type: "geojson",
   302→        data: toFc(GATES.map((g) => polyFeat(circlePolygon(g.position.lat, g.position.lon, 18), { id: g.id }))),
   303→      });
   304→      map.addSource("track", { type: "geojson", data: toFc([]) });
   305→      map.addSource("route-remaining", { type: "geojson", data: toFc([]) });
   306→      map.addSource("route-traveled", { type: "geojson", data: toFc([]) });
   307→
   308→      // --- capas ---
   309→      map.addLayer({ id: "park-fill", type: "fill", source: "park", paint: { "fill-color": "#10b981", "fill-opacity": 0.09 } });
   310→      map.addLayer({ id: "park-line", type: "line", source: "park", paint: { "line-color": "#34d399", "line-width": 1.4, "line-opacity": 0.8 } });
   311→      map.addLayer({ id: "risk-fill", type: "fill", source: "risk", paint: { "fill-color": "#f59e0b", "fill-opacity": 0.14 } });
   312→      map.addLayer({ id: "risk-line", type: "line", source: "risk", paint: { "line-color": "#f59e0b", "line-width": 1, "line-opacity": 0.7 } });
   313→      map.addLayer({ id: "gates-fill", type: "fill", source: "gates", paint: { "fill-color": "#0f766e", "fill-opacity": 0.9 } });
   314→      map.addLayer({
   315→        id: "track-line", type: "line", source: "track",
   316→        layout: { "line-cap": "round", "line-join": "round" },
   317→        paint: { "line-color": "#a3e635", "line-width": 3.5, "line-opacity": 0.9 },
   318→      });
   319→      map.addLayer({
   320→        id: "traveled-line", type: "line", source: "route-traveled",
   321→        filter: ["==", ["get", "ciclo"], true],
   322→        layout: { "line-cap": "round", "line-join": "round" },
   323→        paint: { "line-color": COLOR_CICLO, "line-width": 5, "line-opacity": 0.3 },
   324→      });
   325→      map.addLayer({
   326→        id: "traveled-line-via", type: "line", source: "route-traveled",
   327→        filter: ["==", ["get", "ciclo"], false],
   328→        layout: { "line-cap": "round", "line-join": "round" },
   329→        paint: { "line-color": COLOR_VIA, "line-width": 4, "line-opacity": 0.28 },
   330→      });
   331→      map.addLayer({
   332→        id: "route-casing", type: "line", source: "route-remaining",
   333→        filter: ["==", ["get", "kind"], "casing"],
   334→        layout: { "line-cap": "round", "line-join": "round" },
   335→        paint: { "line-color": COLOR_CASING, "line-width": ["interpolate", ["linear"], ["zoom"], 15, 9, 18, 14], "line-opacity": 0.9 },
   336→      });
   337→      map.addLayer({
   338→        id: "route-ciclo", type: "line", source: "route-remaining",
   339→        filter: ["==", ["get", "ciclo"], true],
   340→        layout: { "line-cap": "round", "line-join": "round" },
   341→        paint: { "line-color": COLOR_CICLO, "line-width": ["interpolate", ["linear"], ["zoom"], 15, 6.5, 18, 10], "line-opacity": 0.98 },
   342→      });
   343→      map.addLayer({
   344→        id: "route-via", type: "line", source: "route-remaining",
   345→        filter: ["all", ["==", ["get", "ciclo"], false], ["==", ["get", "kind"], "seg"]],
   346→        layout: { "line-cap": "round", "line-join": "round" },
   347→        paint: { "line-color": COLOR_VIA, "line-width": ["interpolate", ["linear"], ["zoom"], 15, 5, 18, 8], "line-opacity": 0.98 },
   348→      });
   349→      map.addLayer({
   350→        id: "route-arrows", type: "symbol", source: "route-remaining",
   351→        filter: ["==", ["get", "kind"], "seg"],
   352→        layout: {
   353→          "symbol-placement": "line",
   354→          "symbol-spacing": 130,
   355→          "icon-image": "c30-chev",
   356→          "icon-size": ["interpolate", ["linear"], ["zoom"], 15, 0.35, 18, 0.6],
   357→          "icon-rotation-alignment": "map",
   358→          "icon-allow-overlap": true,
   359→          "icon-ignore-placement": true,
   360→        },
   361→        paint: { "icon-opacity": 0.95 },
   362→      });
   363→
   364→      setReady(true);
   365→    });
   366→
   367→    map.on("click", (e) => {
   368→      if (pickModeRef.current) onPickRef.current({ lat: e.lngLat.lat, lon: e.lngLat.lng });
   369→    });
   370→
   371→    const ro = new ResizeObserver(() => map.resize());
   372→    ro.observe(boxRef.current);
   373→
   374→    return () => {
   375→      ro.disconnect();
   376→      map.remove();
   377→      mapRef.current = null;
   378→    };
   379→  }, []);
   380→
   381→  // ---- marcadores DOM ----
   382→  const puckRef = useRef<maplibregl.Marker | null>(null);
   383→  const arrowRef = useRef<HTMLElement | null>(null);
   384→  const puckPinRef = useRef<string | undefined>(undefined);
   385→  const flagRef = useRef<maplibregl.Marker | null>(null);
   386→  const badgeRef = useRef<maplibregl.Marker | null>(null);
   387→  const badgeIconRef = useRef<string | null>(null);
   388→  const incidentMarkersRef = useRef<maplibregl.Marker[]>([]);
   389→
   390→  // pin del usuario (recrear solo si cambia el pin elegido)
   391→  useEffect(() => {
   392→    const map = mapRef.current;
   393→    if (!map || !ready || !props.position) return;
   394→    if (!puckRef.current) {
   395→      const { el, arrow } = buildPuck(props.userPin);
   396→      puckPinRef.current = props.userPin;
   397→      arrowRef.current = arrow;
   398→      puckRef.current = new maplibregl.Marker({ element: el, rotationAlignment: "map" })
   399→        .setLngLat([props.position.lon, props.position.lat])
   400→        .addTo(map);
   401→    } else if (props.userPin !== puckPinRef.current) {
   402→      const { el, arrow } = buildPuck(props.userPin);
   403→      puckPinRef.current = props.userPin;
   404→      arrowRef.current = arrow;
   405→      puckRef.current.remove();
   406→      puckRef.current = new maplibregl.Marker({ element: el, rotationAlignment: "map" })
   407→        .setLngLat([props.position.lon, props.position.lat])
   408→        .addTo(map);
   409→    }
   410→  }, [ready, props.position, props.userPin]);
   411→
   412→  // posición viva del pin: sigue cada arreglo GPS
   413→  useEffect(() => {
   414→    if (puckRef.current && props.position) {
   415→      puckRef.current.setLngLat([props.position.lon, props.position.lat]);
   416→    }
   417→  }, [props.position]);
   418→
   419→  // bandera de destino + insignia del próximo giro
   420→  useEffect(() => {
   421→    const map = mapRef.current;
   422→    if (!map || !ready) return;
   423→    if (props.dest) {
   424→      if (!flagRef.current) {
   425→        flagRef.current = new maplibregl.Marker({ element: buildFlagEl() }).setLngLat([props.dest.lon, props.dest.lat]).addTo(map);
   426→      } else {
   427→        flagRef.current.setLngLat([props.dest.lon, props.dest.lat]);
   428→      }
   429→    } else if (flagRef.current) {
   430→      flagRef.current.remove();
   431→      flagRef.current = null;
   432→    }
   433→
   434→    const badge = props.nextStep && props.nextStep.icon !== "end" ? props.nextStep : null;
   435→    if (badge) {
   436→      if (!badgeRef.current) {
   437→        badgeRef.current = new maplibregl.Marker({ element: buildBadgeEl(badge.icon) }).setLngLat([badge.point.lon, badge.point.lat]).addTo(map);
   438→        badgeIconRef.current = badge.icon;
   439→      } else {
   440→        badgeRef.current.setLngLat([badge.point.lon, badge.point.lat]);
   441→        if (badgeIconRef.current !== badge.icon) {
   442→          // reemplazo del elemento: el DOM interno cambia de glifo
   443→          const oldEl = badgeRef.current.getElement();
   444→          const newEl = buildBadgeEl(badge.icon);
   445→          if (oldEl.parentNode) oldEl.parentNode.replaceChild(newEl, oldEl);
   446→          badgeIconRef.current = badge.icon;
   447→        }
   448→      }
   449→    } else if (badgeRef.current) {
   450→      badgeRef.current.remove();
   451→      badgeRef.current = null;
   452→      badgeIconRef.current = null;
   453→    }
   454→  }, [ready, props.dest, props.nextStep]);
   455→
   456→  // incidencias de la comunidad
   457→  useEffect(() => {
   458→    const map = mapRef.current;
   459→    if (!map || !ready) return;
   460→    for (const m of incidentMarkersRef.current) m.remove();
   461→    incidentMarkersRef.current = props.incidents.map((inc) =>
   462→      new maplibregl.Marker({ element: buildIncidentEl(inc.type) }).setLngLat([inc.lon, inc.lat]).addTo(map)
   463→    );
   464→  }, [ready, props.incidents]);
   465→
   466→  // ---- datos GeoJSON ----
   467→  useEffect(() => {
   468→    const map = mapRef.current;
   469→    if (!map || !ready) return;
   470→
   471→    // recorrido del ride
   472→    const trackSrc = map.getSource("track") as maplibregl.GeoJSONSource | undefined;
   473→    trackSrc?.setData(props.track.length > 1 ? toFc([lineFeat(props.track)]) : toFc([]));
   474→
   475→    if (!props.onRoute || !props.route) {
   476→      (map.getSource("route-remaining") as maplibregl.GeoJSONSource | undefined)?.setData(toFc([]));
   477→      (map.getSource("route-traveled") as maplibregl.GeoJSONSource | undefined)?.setData(toFc([]));
   478→      return;
   479→    }
   480→
   481→    const r = props.route;
   482→    // El tramo restante SIEMPRE parte de la ubicación ACTUAL: si el
   483→    // pin ya se movió del vértice más cercano, se antepone la
   484→    // posición viva como origen con un conector directo (así la ruta
   485→    // nace en tu punto, no en la ciclovía más próxima).
   486→    const feats = buildRouteFeatures(r, props.nearestIdx, props.position);
   487→
   488→    (map.getSource("route-remaining") as maplibregl.GeoJSONSource | undefined)?.setData(feats.remaining);
   489→    (map.getSource("route-traveled") as maplibregl.GeoJSONSource | undefined)?.setData(feats.traveled);
   490→  }, [ready, props.onRoute, props.route, props.nearestIdx, props.position, props.track]);
   491→
   492→  // ---- cámara: pin FIJO abajo y al centro, estilo Waze ----
   493→  useEffect(() => {
   494→    const map = mapRef.current;
   495→    if (!map || !ready) return;
   496→
   497→    // Sin posición: encuadra la ruta completa (vista previa plana)
   498→    if (!props.position) {
   499→      if (props.onRoute && props.route) {
   500→        const lons = props.route.coords.map((c) => c.lon);
   501→        const lats = props.route.coords.map((c) => c.lat);
   502→        map.easeTo({
   503→          center: [(Math.min(...lons) + Math.max(...lons)) / 2, (Math.min(...lats) + Math.max(...lats)) / 2],
   504→          zoom: 13.8,
   505→          pitch: 0,
   506→          bearing: 0,
   507→          duration: 700,
   508→        });
   509→      }
   510→      return;
   511→    }
   512→
   513→    // El CENTRO de la cámara es la posición del ciclista y el relleno
   514→    // superior corre el punto de ancla al ~30% del alto: el pin queda
   515→    // clavado abajo y centrado mientras el mundo fluye bajo él.
   516→    const h = boxRef.current?.clientHeight ?? 260;
   517→    const anchorBottom = Math.round(h * (props.onRoute ? 0.3 : 0.34));
   518→    const padding = { top: Math.max(0, Math.round(h) - anchorBottom * 2), bottom: 0, left: 0, right: 0 };
   519→
   520→    // rumbo: sensor de curso → dirección de la ruta → mantener actual
   521→    let b: number;
   522→    if (props.heading !== null) {
   523→      b = props.heading;
   524→    } else if (props.onRoute && props.route && props.nearestIdx !== null && props.nearestIdx < props.route.coords.length - 1) {
   525→      b = bearingOf(props.route.coords[props.nearestIdx], props.route.coords[props.nearestIdx + 1]);
   526→    } else {
   527→      b = map.getBearing();
   528→    }
   529→    const cur = map.getBearing();
   530→    // giro por el camino corto (evita vueltas de 340°)
   531→    while (b - cur > 180) b -= 360;
   532→    while (b - cur < -180) b += 360;
   533→
   534→    const camZoom = props.onRoute && props.remainingM < ARRIVAL_M ? ARRIVAL_ZOOM : NAV_ZOOM;
   535→
   536→    if (!firstFixRef.current) {
   537→      firstFixRef.current = true;
   538→      map.jumpTo({
   539→        center: [props.position.lon, props.position.lat],
   540→        bearing: b,
   541→        pitch: NAV_PITCH,
   542→        zoom: camZoom,
   543→        padding,
   544→      });
   545→      return;
   546→    }
   547→
   548→    map.easeTo({
   549→      center: [props.position.lon, props.position.lat],
   550→      bearing: b,
   551→      pitch: NAV_PITCH,
   552→      zoom: camZoom,
   553→      padding,
   554→      duration: 680,
   555→      easing: (t) => t,
   556→    });
   557→  }, [ready, props.position, props.heading, props.onRoute, props.remainingM, props.nearestIdx, props.route]);
   558→
   559→  // rotación del pin flecha respecto al mapa (queda alineado al rumbo)
   560→  useEffect(() => {
   561→    const map = mapRef.current;
   562→    if (!map || !ready) return;
   563→    const apply = () => {
   564→      const m = puckRef.current;
   565→      const arrow = arrowRef.current;
   566→      if (!m || !arrow) return;
   567→      const hdg = props.heading ?? 0;
   568→      arrow.style.transform = `rotate(${hdg - map.getBearing()}deg)`;
   569→    };
   570→    apply();
   571→    map.on("rotate", apply);
   572→    return () => {
   573→      map.off("rotate", apply);
   574→    };
   575→  }, [ready, props.heading, props.position]);
   576→
   577→  // vista a un punto (localización sin ride)
   578→  useEffect(() => {
   579→    const map = mapRef.current;
   580→    if (!map || !ready || !props.locateTarget) return;
   581→    map.flyTo({ center: [props.locateTarget.lon, props.locateTarget.lat], zoom: Math.max(map.getZoom(), 16), duration: 800 });
   582→  }, [ready, props.locateTarget]);
   583→
   584→  return (
   585→    // position absolute por estilo inline: el CSS de maplibre usa
   586→    // .maplibregl-map{position:relative} y pisa la utilidad `absolute`
   587→    <div ref={boxRef} style={{ position: "absolute", inset: 0 }} />
   588→  );
   589→}

   789→// ============================================================
   790→// Componente principal: lienzo + capa Waze (banner, barra, etc.)
   791→// ============================================================
   792→export default function MiniMap({ position, heading, track, accuracyM, userPin, nav, speedKmh = 0 }: Props) {
   793→  const incidents = useIncidentFeed();
   794→  const [pickMode, setPickMode] = useState(false);
   795→  const [locateTarget, setLocateTarget] = useState<LatLng | null>(null);
   796→  const [reportOpen, setReportOpen] = useState(false);
   797→  const [status, setStatus] = useState<string | null>(null);
   798→  const route = nav.route;
   799→  const dest = nav.dest;
   800→  // Mapa nativo (MapLibre Native vía plugin) o lienzo web (MapLibre GL JS)
   801→  // ⚠️ NATIVE_MAPS_ENABLED=false (v1.3.1): el WebView del dispositivo no
   802→  // renderiza ni el nativo ni MapLibre GL JS (WebGL caído) — se muestra un
   803→  // fondo de color sin mapa. Prioriza funcionamiento: en la app nativa
   804→  // corre Leaflet con tiles ráster (LeafletNav); el código nativo queda
   805→  // dormido detrás de este flag para depurarlo más adelante.
   806→  const NATIVE_MAPS_ENABLED = false;
   807→  const nativeDetected = useMemo(() => isNativeMapPlatform(), []);
   808→  const [nativeFail, setNativeFail] = useState(false);
   809→  const native = NATIVE_MAPS_ENABLED && nativeDetected && !nativeFail;
   810→
   811→  const webglOk = useMemo(() => {
   812→    try {
   813→      return webglSupported();
   814→    } catch {
   815→      return false;
   816→    }
   817→  }, []);
   818→
   819→  const pick = (p: LatLng) => {
   820→    nav.setDestination({ name: "Punto en el mapa", position: p });
   821→    setPickMode(false);
   822→  };
   823→
   824→  const locate = () => {
   825→    if (position) {
   826→      setLocateTarget(position);
   827→      return;
   828→    }
   829→    if ("geolocation" in navigator) {
   830→      navigator.geolocation.getCurrentPosition(
   831→        (pos) => setLocateTarget({ lat: pos.coords.latitude, lon: pos.coords.longitude }),
   832→        () => {},
   833→        { enableHighAccuracy: true, timeout: 8000 }
   834→      );
   835→    }
   836→  };
   837→
   838→  const onRoute = !!route && route.coords.length > 1;
   839→
   840→  const progress = useMemo(
   841→    () => (onRoute && position ? routeProgress(route!, position) : null),
   842→    [onRoute, route, position]
   843→  );
   844→
   845→  const nextStep = progress?.nextStep ?? route?.steps[0] ?? null;
   846→  const laterStep = onRoute && progress && route ? route.steps[progress.passedCount + 1] ?? null : null;
   847→  const offRoute = !!progress && progress.offRouteM > 140;
   848→  const remainingM = progress?.remainingM ?? route?.distanceM ?? 0;
   849→
   850→  const currentStreet = onRoute && progress ? streetAt(route!, progress.nearestIdx) : "";
   851→
   852→  const spd = speedKmh > 4 ? speedKmh : 15;
   853→  const etaMin = remainingM > 0 ? Math.max(1, Math.round((remainingM / 1000 / spd) * 60)) : 0;
   854→  const arrivalClock = useMemo(
   855→    () => new Date(Date.now() + etaMin * 60000).toLocaleTimeString("es-CL", { hour: "2-digit", minute: "2-digit", hourCycle: "h23" }),
   856→    [etaMin]
   857→  );
   858→
   859→  // ---- Reporte rápido de incidencias (botón amarillo) ----
   860→  const reportIncident = async (type: IncidentTypeId) => {
   861→    if (!position) {
   862→      setStatus("Aún sin GPS: no podemos fijar la ubicación del reporte.");
   863→      setReportOpen(false);
   864→      setTimeout(() => setStatus(null), 4000);
   865→      return;
   866→    }
   867→    setReportOpen(false);
   868→    const payload = { type, lat: position.lat, lon: position.lon };
   869→    try {
   870→      const r = await fetch("/api/incidents", {
   871→        method: "POST",
   872→        headers: { "Content-Type": "application/json" },
   873→        body: JSON.stringify(payload),
   874→      });
   875→      const d = (await r.json()) as { incident?: IncidentItem; error?: string };
   876→      if (!r.ok || !d.incident) throw new Error(d.error ?? "error");
   877→      addIncident(d.incident);
   878→      setStatus("Incidencia reportada ✓ visible para toda la comunidad");
   879→    } catch {
   880→      const local: IncidentItem = {
   881→        id: crypto.randomUUID(),
   882→        ...payload,
   883→        note: null,
   884→        createdAt: new Date().toISOString(),
   885→        username: "tú (local)",
   886→        authorPin: userPin ?? DEFAULT_PIN,
   887→        mine: true,
   888→        localOnly: true,
   889→      };
   890→      addIncident(local);
   891→      saveLocalIncidents([local, ...loadLocalIncidents()]);
   892→      setStatus("Sin conexión: reporte guardado en este dispositivo.");
   893→    }
   894→    setTimeout(() => setStatus(null), 5000);
   895→  };
   896→
   897→  // ---- ramas de render: NATIVO (MapLibre Native) o WEB (MapLibre GL JS) ----
   898→  if (native) {
   899→    return (
   900→      <div className="flex flex-col gap-1.5">
   901→        {/* Banner de maniobra ENCIMA del lienzo nativo */}
