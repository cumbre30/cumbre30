    79→    79→/** Rumbo (0-360) entre dos puntos. */
    80→    80→function bearingOf(a: LatLng, b: LatLng): number {
    81→    81→  const toR = Math.PI / 180;
    82→    82→  const y = Math.sin((b.lon - a.lon) * toR) * Math.cos(b.lat * toR);
    83→    83→  const x = Math.cos(a.lat * toR) * Math.sin(b.lat * toR) - Math.sin(a.lat * toR) * Math.cos(b.lat * toR) * Math.cos((b.lon - a.lon) * toR);
    84→    84→  return (Math.atan2(y, x) / toR + 360) % 360;
    85→    85→}

    88→    88→function offsetAlong(p: LatLng, headingDeg: number, distM: number): LatLng {
    89→    89→  const r = (headingDeg * Math.PI) / 180;
    90→    90→  const mPerDegLat = 111_320;
    91→    91→  const mPerDegLon = 111_320 * Math.cos((p.lat * Math.PI) / 180);
    92→    92→  return {
    93→    93→    lat: p.lat + (Math.cos(r) * distM) / mPerDegLat,
    94→    94→    lon: p.lon + (Math.sin(r) * distM) / mPerDegLon,
    95→    95→  };
    96→    96→}

   539→   539→    while (b - cur < -180) b += 360;
   540→   540→

   550→   550→        padding: { top: 0, bottom: Math.round(h * 0.3), left: 0, right: 0 },
   551→   551→      });
   552→   552→      return;
   553→   553→    }
   554→   554→
   555→   555→    map.easeTo({
   556→   556→      center: [target.lon, target.lat],
   557→   557→      bearing: b,
   558→   558→      pitch: NAV_PITCH,
   559→   559→      zoom: camZoom,
   560→   560→      padding: { top: 0, bottom: Math.round(h * 0.3), left: 0, right: 0 },
   561→   561→      duration: 680,
   562→   562→      easing: (t) => t,
   563→   563→    });
   564→   564→  }, [ready, props.position, props.heading, props.onRoute, props.offRoute, props.remainingM, props.nearestIdx, props.route]);

   580→   580→      map.off("rotate", apply);
   581→   581→    };
   582→   582→  }, [ready, props.heading, props.position]);

   589→   589→  }, [ready, props.locateTarget]);
   590→   590→
   591→   591→  return (
   592→   592→    // position absolute por estilo inline: el CSS de maplibre usa

   595→   595→  );
   596→   596→}

   602→   602→  const incidents = useIncidentFeed();

   605→   605→  const [reportOpen, setReportOpen] = useState(false);

   724→   724→      {/* ===== Banner de maniobra (modo ruta), estilo Waze ===== */}

   729→   729→            <div className="font-mono text-[26px] font-bold text-zinc-50 leading-none tabular-nums">
   730→   730→              {nextStep.icon === "end" ? "Llegaste" : fmtKmM(progress?.distToNextStepM ?? nextStep.distM)}
   731→   731→            </div>
   732→   732→            <div className="text-lg font-bold text-sky-400 leading-tight truncate mt-0.5">
   733→   733→              {nextStep.icon === "end" ? dest?.name ?? "Destino" : streetOfStep(nextStep.text)}
   734→   734→            </div>
   735→   735→            {offRoute && <div className="text-[11px] text-amber-300 font-semibold mt-0.5">desviado, recalculando…</div>}
   736→   736→          </div>
   737→   737→          {laterStep && laterStep.icon !== "end" && <TurnArrow icon={laterStep.icon} className="h-7 w-7 shrink-0 mt-1 text-zinc-500" />}
   738→   738→        </div>
   739→   739→      )}
   740→   740→
   741→   741→      {/* Estado de cálculo de ruta */}
   742→   742→      {nav.loading && (
   743→   743→        <div className="absolute top-2 left-2 z-[520] rounded-full bg-zinc-950/90 border border-sky-700 px-2.5 py-1 text-[10px] font-semibold text-sky-300 flex items-center gap-1.5 pointer-events-none">

   748→   748→
   749→   749→      {/* Leyenda compacta de la ruta (solo modo ruta) */}
   750→   750→      {onRoute && !nav.loading && (
