   140→  ctx.strokeStyle = "#ffffff";

   225→function NavCanvas(props: NavCanvasProps) {
   226→  const boxRef = useRef<HTMLDivElement>(null);
   227→  const mapRef = useRef<MlMap | null>(null);
   228→  const [ready, setReady] = useState(false);
   229→  const firstFixRef = useRef(false);
   230→
   231→  // refs vivas para handlers registrados una sola vez
   232→  const pickModeRef = useRef(props.pickMode);
   233→  const onPickRef = useRef(props.onPick);
   234→  useEffect(() => {
   235→    pickModeRef.current = props.pickMode;
   236→    onPickRef.current = props.onPick;
   237→  }, [props.pickMode, props.onPick]);
   238→
   239→  // ---- init ----
   240→  useEffect(() => {
   241→    if (!boxRef.current || mapRef.current) return;
   242→    const map = new maplibregl.Map({
   243→      container: boxRef.current,
   244→      style: navStyle(),
   245→      center: PARK_CENTER,
   246→      zoom: 13.5,
   247→      pitch: 0,
   248→      bearing: 0,
   249→      attributionControl: false,
   250→      dragRotate: false,
   251→      dragPan: false,
   252→      scrollZoom: false,
   253→      doubleClickZoom: false,
   254→      touchZoomRotate: false,
   255→      keyboard: false,
   256→      boxZoom: false,
   257→      maxPitch: 70,
   258→    });
   259→    mapRef.current = map;
   260→    map.on("load", () => {
   261→      map.addImage("c30-chev", makeChevronImage());
   262→
   263→      // asa de depuración/E2E (no interfiere en producción)
   264→      (window as unknown as Record<string, unknown>).__c30nav = map;
   265→
   266→      // --- fuentes ---
   267→      map.addSource("park", { type: "geojson", data: toFc([polyFeat(PARK_POLYGON.map((p) => [p.lon, p.lat] as [number, number]))]) });
   268→      map.addSource("risk", {
   269→        type: "geojson",
   270→        data: toFc(RISK_ZONES.map((z) => polyFeat(circlePolygon(z.position.lat, z.position.lon, z.radiusM), { id: z.id }))),
   271→      });
   272→      map.addSource("gates", {
   273→        type: "geojson",
   274→        data: toFc(GATES.map((g) => polyFeat(circlePolygon(g.position.lat, g.position.lon, 18), { id: g.id }))),
   275→      });
   276→      map.addSource("track", { type: "geojson", data: toFc([]) });
   277→      map.addSource("route-remaining", { type: "geojson", data: toFc([]) });
   278→      map.addSource("route-traveled", { type: "geojson", data: toFc([]) });
   279→
   280→      // --- capas ---
   281→      map.addLayer({ id: "park-fill", type: "fill", source: "park", paint: { "fill-color": "#10b981", "fill-opacity": 0.09 } });
   282→      map.addLayer({ id: "park-line", type: "line", source: "park", paint: { "line-color": "#34d399", "line-width": 1.4, "line-opacity": 0.8 } });
   283→      map.addLayer({ id: "risk-fill", type: "fill", source: "risk", paint: { "fill-color": "#f59e0b", "fill-opacity": 0.14 } });
   284→      map.addLayer({ id: "risk-line", type: "line", source: "risk", paint: { "line-color": "#f59e0b", "line-width": 1, "line-opacity": 0.7 } });
   285→      map.addLayer({ id: "gates-fill", type: "fill", source: "gates", paint: { "fill-color": "#0f766e", "fill-opacity": 0.9 } });
   286→      map.addLayer({
   287→        id: "track-line", type: "line", source: "track",
   288→        layout: { "line-cap": "round", "line-join": "round" },
   289→        paint: { "line-color": "#a3e635", "line-width": 3.5, "line-opacity": 0.9 },
   290→      });
   291→      map.addLayer({
   292→        id: "traveled-line", type: "line", source: "route-traveled",
   293→        filter: ["==", ["get", "ciclo"], true],
   294→        layout: { "line-cap": "round", "line-join": "round" },
   295→        paint: { "line-color": COLOR_CICLO, "line-width": 5, "line-opacity": 0.3 },
   296→      });
   297→      map.addLayer({
   298→        id: "traveled-line-via", type: "line", source: "route-traveled",
   299→        filter: ["==", ["get", "ciclo"], false],
   300→        layout: { "line-cap": "round", "line-join": "round" },
   301→        paint: { "line-color": COLOR_VIA, "line-width": 4, "line-opacity": 0.28 },
   302→      });
   303→      map.addLayer({
   304→        id: "route-casing", type: "line", source: "route-remaining",
   305→        filter: ["==", ["get", "kind"], "casing"],
   306→        layout: { "line-cap": "round", "line-join": "round" },
   307→        paint: { "line-color": COLOR_CASING, "line-width": ["interpolate", ["linear"], ["zoom"], 15, 9, 18, 14], "line-opacity": 0.9 },
   308→      });
   309→      map.addLayer({
   310→        id: "route-ciclo", type: "line", source: "route-remaining",
   311→        filter: ["==", ["get", "ciclo"], true],
   312→        layout: { "line-cap": "round", "line-join": "round" },
   313→        paint: { "line-color": COLOR_CICLO, "line-width": ["interpolate", ["linear"], ["zoom"], 15, 6.5, 18, 10], "line-opacity": 0.98 },
   314→      });
   315→      map.addLayer({
   316→        id: "route-via", type: "line", source: "route-remaining",
   317→        filter: ["all", ["==", ["get", "ciclo"], false], ["==", ["get", "kind"], "seg"]],
   318→        layout: { "line-cap": "round", "line-join": "round" },
   319→        paint: { "line-color": COLOR_VIA, "line-width": ["interpolate", ["linear"], ["zoom"], 15, 5, 18, 8], "line-opacity": 0.98 },
   320→      });
   321→      map.addLayer({
   322→        id: "route-arrows", type: "symbol", source: "route-remaining",
   323→        filter: ["==", ["get", "kind"], "seg"],
   324→        layout: {
   325→          "symbol-placement": "line",
   326→          "symbol-spacing": 130,
   327→          "icon-image": "c30-chev",
   328→          "icon-size": ["interpolate", ["linear"], ["zoom"], 15, 0.35, 18, 0.6],
   329→          "icon-rotation-alignment": "map",
   330→          "icon-allow-overlap": true,
   331→          "icon-ignore-placement": true,
   332→        },
   333→        paint: { "icon-opacity": 0.95 },
   334→      });
   335→
   336→      setReady(true);
   337→    });
   338→
   339→    map.on("click", (e) => {
   340→      if (pickModeRef.current) onPickRef.current({ lat: e.lngLat.lat, lon: e.lngLat.lng });
   341→    });
   342→
   343→    const ro = new ResizeObserver(() => map.resize());
   344→    ro.observe(boxRef.current);
   345→
   346→    return () => {
   347→      ro.disconnect();
   348→      map.remove();
   349→      mapRef.current = null;
   350→    };
   351→  }, []);
   352→
   353→  // ---- marcadores DOM ----
   354→  const puckRef = useRef<maplibregl.Marker | null>(null);
   355→  const arrowRef = useRef<HTMLElement | null>(null);
   356→  const puckPinRef = useRef<string | undefined>(undefined);
   357→  const flagRef = useRef<maplibregl.Marker | null>(null);
   358→  const badgeRef = useRef<maplibregl.Marker | null>(null);
   359→  const badgeIconRef = useRef<string | null>(null);
   360→  const incidentMarkersRef = useRef<maplibregl.Marker[]>([]);
   361→
   362→  // pin del usuario (recrear solo si cambia el pin elegido)
   363→  useEffect(() => {
   364→    const map = mapRef.current;
   365→    if (!map || !ready || !props.position) return;
   366→    if (!puckRef.current) {
   367→      const { el, arrow } = buildPuck(props.userPin);
   368→      puckPinRef.current = props.userPin;
   369→      arrowRef.current = arrow;
   370→      puckRef.current = new maplibregl.Marker({ element: el, rotationAlignment: "map" })
   371→        .setLngLat([props.position.lon, props.position.lat])
   372→        .addTo(map);
   373→    } else if (props.userPin !== puckPinRef.current) {
   374→      const { el, arrow } = buildPuck(props.userPin);
   375→      puckPinRef.current = props.userPin;
   376→      arrowRef.current = arrow;
   377→      puckRef.current.remove();
   378→      puckRef.current = new maplibregl.Marker({ element: el, rotationAlignment: "map" })
   379→        .setLngLat([props.position.lon, props.position.lat])
   380→        .addTo(map);
   381→    }
   382→  }, [ready, props.position, props.userPin]);
   383→
   384→  // posición viva del pin: sigue cada arreglo GPS
   385→  useEffect(() => {
   386→    if (puckRef.current && props.position) {
   387→      puckRef.current.setLngLat([props.position.lon, props.position.lat]);
   388→    }
   389→  }, [props.position]);
   390→
   391→  // bandera de destino + insignia del próximo giro
   392→  useEffect(() => {
   393→    const map = mapRef.current;
   394→    if (!map || !ready) return;
   395→    if (props.dest) {
   396→      if (!flagRef.current) {
   397→        flagRef.current = new maplibregl.Marker({ element: buildFlagEl() }).setLngLat([props.dest.lon, props.dest.lat]).addTo(map);
   398→      } else {
   399→        flagRef.current.setLngLat([props.dest.lon, props.dest.lat]);
   400→      }
   401→    } else if (flagRef.current) {
   402→      flagRef.current.remove();
   403→      flagRef.current = null;
   404→    }
   405→
   406→    const badge = props.nextStep && props.nextStep.icon !== "end" ? props.nextStep : null;
   407→    if (badge) {
   408→      if (!badgeRef.current) {
   409→        badgeRef.current = new maplibregl.Marker({ element: buildBadgeEl(badge.icon) }).setLngLat([badge.point.lon, badge.point.lat]).addTo(map);
   410→        badgeIconRef.current = badge.icon;
   411→      } else {
   412→        badgeRef.current.setLngLat([badge.point.lon, badge.point.lat]);
   413→        if (badgeIconRef.current !== badge.icon) {
   414→          // reemplazo del elemento: el DOM interno cambia de glifo
   415→          const oldEl = badgeRef.current.getElement();
   416→          const newEl = buildBadgeEl(badge.icon);
   417→          if (oldEl.parentNode) oldEl.parentNode.replaceChild(newEl, oldEl);
   418→          badgeIconRef.current = badge.icon;
   419→        }
   420→      }
   421→    } else if (badgeRef.current) {
   422→      badgeRef.current.remove();
   423→      badgeRef.current = null;
   424→      badgeIconRef.current = null;
   425→    }
   426→  }, [ready, props.dest, props.nextStep]);
   427→
   428→  // incidencias de la comunidad
   429→  useEffect(() => {
   430→    const map = mapRef.current;
   431→    if (!map || !ready) return;
   432→    for (const m of incidentMarkersRef.current) m.remove();
   433→    incidentMarkersRef.current = props.incidents.map((inc) =>
   434→      new maplibregl.Marker({ element: buildIncidentEl(inc.type) }).setLngLat([inc.lon, inc.lat]).addTo(map)
   435→    );
   436→  }, [ready, props.incidents]);
   437→
   438→  // ---- datos GeoJSON ----
   439→  useEffect(() => {
   440→    const map = mapRef.current;
   441→    if (!map || !ready) return;
   442→
   443→    // recorrido del ride
   444→    const trackSrc = map.getSource("track") as maplibregl.GeoJSONSource | undefined;
   445→    trackSrc?.setData(props.track.length > 1 ? toFc([lineFeat(props.track)]) : toFc([]));
   446→
   447→    if (!props.onRoute || !props.route) {
   448→      (map.getSource("route-remaining") as maplibregl.GeoJSONSource | undefined)?.setData(toFc([]));
   449→      (map.getSource("route-traveled") as maplibregl.GeoJSONSource | undefined)?.setData(toFc([]));
   450→      return;
   451→    }
   452→
   453→    const r = props.route;
   454→    // El tramo restante SIEMPRE parte de la ubicación ACTUAL: si el
   455→    // pin ya se movió del vértice más cercano, se antepone la
   456→    // posición viva como origen con un conector directo (así la ruta
   457→    // nace en tu punto, no en la ciclovía más próxima).
   458→    const feats = buildRouteFeatures(r, props.nearestIdx, props.position);
   459→
   460→    (map.getSource("route-remaining") as maplibregl.GeoJSONSource | undefined)?.setData(feats.remaining);
   461→    (map.getSource("route-traveled") as maplibregl.GeoJSONSource | undefined)?.setData(feats.traveled);
   462→  }, [ready, props.onRoute, props.route, props.nearestIdx, props.position, props.track]);
   463→
   464→  // ---- cámara: pin FIJO abajo y al centro, estilo Waze ----
   465→  useEffect(() => {
   466→    const map = mapRef.current;
   467→    if (!map || !ready) return;
   468→
   469→    // Sin posición: encuadra la ruta completa (vista previa plana)
   470→    if (!props.position) {
   471→      if (props.onRoute && props.route) {
   472→        const lons = props.route.coords.map((c) => c.lon);
   473→        const lats = props.route.coords.map((c) => c.lat);
   474→        map.easeTo({
   475→          center: [(Math.min(...lons) + Math.max(...lons)) / 2, (Math.min(...lats) + Math.max(...lats)) / 2],
   476→          zoom: 13.8,
   477→          pitch: 0,
   478→          bearing: 0,
   479→          duration: 700,
   480→        });
   481→      }
   482→      return;
   483→    }
   484→
   485→    // El CENTRO de la cámara es la posición del ciclista y el relleno
   486→    // superior corre el punto de ancla al ~30% del alto: el pin queda
   487→    // clavado abajo y centrado mientras el mundo fluye bajo él.
   488→    const h = boxRef.current?.clientHeight ?? 260;
   489→    const anchorBottom = Math.round(h * (props.onRoute ? 0.3 : 0.34));
   490→    const padding = { top: Math.max(0, Math.round(h) - anchorBottom * 2), bottom: 0, left: 0, right: 0 };
   491→
   492→    // rumbo: sensor de curso → dirección de la ruta → mantener actual
   493→    let b: number;
   494→    if (props.heading !== null) {
   495→      b = props.heading;
   496→    } else if (props.onRoute && props.route && props.nearestIdx !== null && props.nearestIdx < props.route.coords.length - 1) {
   497→      b = bearingOf(props.route.coords[props.nearestIdx], props.route.coords[props.nearestIdx + 1]);
   498→    } else {
   499→      b = map.getBearing();
   500→    }
   501→    const cur = map.getBearing();
   502→    // giro por el camino corto (evita vueltas de 340°)
   503→    while (b - cur > 180) b -= 360;
   504→    while (b - cur < -180) b += 360;
   505→
   506→    const camZoom = props.onRoute && props.remainingM < ARRIVAL_M ? ARRIVAL_ZOOM : NAV_ZOOM;
   507→
   508→    if (!firstFixRef.current) {
   509→      firstFixRef.current = true;
   510→      map.jumpTo({
   511→        center: [props.position.lon, props.position.lat],
   512→        bearing: b,
   513→        pitch: NAV_PITCH,
   514→        zoom: camZoom,
   515→        padding,
   516→      });
   517→      return;
   518→    }
   519→
   520→    map.easeTo({
   521→      center: [props.position.lon, props.position.lat],
   522→      bearing: b,
   523→      pitch: NAV_PITCH,
   524→      zoom: camZoom,
   525→      padding,
   526→      duration: 680,
   527→      easing: (t) => t,
   528→    });
   529→  }, [ready, props.position, props.heading, props.onRoute, props.remainingM, props.nearestIdx, props.route]);
   530→
   531→  // rotación del pin flecha respecto al mapa (queda alineado al rumbo)
   532→  useEffect(() => {
   533→    const map = mapRef.current;
   534→    if (!map || !ready) return;
   535→    const apply = () => {
   536→      const m = puckRef.current;
   537→      const arrow = arrowRef.current;
   538→      if (!m || !arrow) return;
   539→      const hdg = props.heading ?? 0;
   540→      arrow.style.transform = `rotate(${hdg - map.getBearing()}deg)`;
   541→    };
   542→    apply();
   543→    map.on("rotate", apply);
   544→    return () => {
   545→      map.off("rotate", apply);
   546→    };
   547→  }, [ready, props.heading, props.position]);
   548→
   549→  // vista a un punto (localización sin ride)
   550→  useEffect(() => {
   551→    const map = mapRef.current;
   552→    if (!map || !ready || !props.locateTarget) return;
   553→    map.flyTo({ center: [props.locateTarget.lon, props.locateTarget.lat], zoom: Math.max(map.getZoom(), 16), duration: 800 });
   554→  }, [ready, props.locateTarget]);
   555→
   556→  return (
   557→    // position absolute por estilo inline: el CSS de maplibre usa
   558→    // .maplibregl-map{position:relative} y pisa la utilidad `absolute`
   559→    <div ref={boxRef} style={{ position: "absolute", inset: 0 }} />
   560→  );
   561→}

   761→// ============================================================
   762→// Componente principal: lienzo + capa Waze (banner, barra, etc.)
   763→// ============================================================
   764→export default function MiniMap({ position, heading, track, accuracyM, userPin, nav, speedKmh = 0 }: Props) {
   765→  const incidents = useIncidentFeed();
   766→  const [pickMode, setPickMode] = useState(false);
   767→  const [locateTarget, setLocateTarget] = useState<LatLng | null>(null);
   768→  const [reportOpen, setReportOpen] = useState(false);
   769→  const [status, setStatus] = useState<string | null>(null);
   770→  const route = nav.route;
   771→  const dest = nav.dest;
   772→  // Mapa nativo (MapLibre Native vía plugin) o lienzo web (MapLibre GL JS)
   773→  const nativeDetected = useMemo(() => isNativeMapPlatform(), []);
   774→  const [nativeFail, setNativeFail] = useState(false);
   775→  const native = nativeDetected && !nativeFail;
   776→
   777→  const webglOk = useMemo(() => {
   778→    try {
   779→      return webglSupported();
   780→    } catch {
   781→      return false;
   782→    }
   783→  }, []);
   784→
   785→  const pick = (p: LatLng) => {
   786→    nav.setDestination({ name: "Punto en el mapa", position: p });
   787→    setPickMode(false);
   788→  };
   789→
   790→  const locate = () => {
   791→    if (position) {
   792→      setLocateTarget(position);
   793→      return;
   794→    }
   795→    if ("geolocation" in navigator) {
   796→      navigator.geolocation.getCurrentPosition(
   797→        (pos) => setLocateTarget({ lat: pos.coords.latitude, lon: pos.coords.longitude }),
   798→        () => {},
   799→        { enableHighAccuracy: true, timeout: 8000 }
   800→      );
   801→    }
   802→  };
   803→
   804→  const onRoute = !!route && route.coords.length > 1;
   805→
   806→  const progress = useMemo(
   807→    () => (onRoute && position ? routeProgress(route!, position) : null),
   808→    [onRoute, route, position]
   809→  );
   810→
   811→  const nextStep = progress?.nextStep ?? route?.steps[0] ?? null;
   812→  const laterStep = onRoute && progress && route ? route.steps[progress.passedCount + 1] ?? null : null;
   813→  const offRoute = !!progress && progress.offRouteM > 140;
   814→  const remainingM = progress?.remainingM ?? route?.distanceM ?? 0;
   815→
   816→  const currentStreet = onRoute && progress ? streetAt(route!, progress.nearestIdx) : "";
   817→
   818→  const spd = speedKmh > 4 ? speedKmh : 15;
   819→  const etaMin = remainingM > 0 ? Math.max(1, Math.round((remainingM / 1000 / spd) * 60)) : 0;
   820→  const arrivalClock = useMemo(
   821→    () => new Date(Date.now() + etaMin * 60000).toLocaleTimeString("es-CL", { hour: "2-digit", minute: "2-digit", hourCycle: "h23" }),
   822→    [etaMin]
   823→  );
   824→
   825→  // ---- Reporte rápido de incidencias (botón amarillo) ----
   826→  const reportIncident = async (type: IncidentTypeId) => {
   827→    if (!position) {
   828→      setStatus("Aún sin GPS: no podemos fijar la ubicación del reporte.");
   829→      setReportOpen(false);
   830→      setTimeout(() => setStatus(null), 4000);
   831→      return;
   832→    }
   833→    setReportOpen(false);
   834→    const payload = { type, lat: position.lat, lon: position.lon };
   835→    try {
   836→      const r = await fetch("/api/incidents", {
   837→        method: "POST",
   838→        headers: { "Content-Type": "application/json" },
   839→        body: JSON.stringify(payload),
   840→      });
   841→      const d = (await r.json()) as { incident?: IncidentItem; error?: string };
   842→      if (!r.ok || !d.incident) throw new Error(d.error ?? "error");
   843→      addIncident(d.incident);
   844→      setStatus("Incidencia reportada ✓ visible para toda la comunidad");
   845→    } catch {
   846→      const local: IncidentItem = {
   847→        id: crypto.randomUUID(),
   848→        ...payload,
   849→        note: null,
   850→        createdAt: new Date().toISOString(),
   851→        username: "tú (local)",
   852→        authorPin: userPin ?? DEFAULT_PIN,
   853→        mine: true,
   854→        localOnly: true,
   855→      };
   856→      addIncident(local);
   857→      saveLocalIncidents([local, ...loadLocalIncidents()]);
   858→      setStatus("Sin conexión: reporte guardado en este dispositivo.");
   859→    }
   860→    setTimeout(() => setStatus(null), 5000);
   861→  };
   862→
   863→  // ---- ramas de render: NATIVO (MapLibre Native) o WEB (MapLibre GL JS) ----
   864→  if (native) {
   865→    return (
   866→      <div className="flex flex-col gap-1.5">
   867→        {/* Banner de maniobra ENCIMA del lienzo nativo */}
   868→        {onRoute && nextStep && (
   869→          <ManeuverBanner
   870→            className="rounded-xl border border-zinc-800 bg-zinc-950/95 px-3 py-2 flex items-center gap-3"
   871→            nextStep={nextStep}
   872→            distM={progress?.distToNextStepM ?? nextStep.distM}
   873→            destName={dest?.name ?? "Destino"}
   874→            offRoute={offRoute}
   875→            laterStep={laterStep}
   876→          />
   877→        )}
   878→
   879→        {/* SOLO el lienzo nativo dentro de la ventana: sin HTML encima */}
   880→        <div className={`relative rounded-xl overflow-hidden border border-zinc-800 shrink-0 ${onRoute ? "h-72" : "h-56"}`}>
   881→          <NativeMapCanvas
   882→            id="minimap"
   883→            mode="nav"
   884→            position={position}
   885→            heading={heading}
   886→            track={track}
   887→            accuracyM={accuracyM}
   888→            onRoute={onRoute}
   889→            route={onRoute ? route : null}
   890→            nearestIdx={progress ? progress.nearestIdx : null}
   891→            remainingM={remainingM}
   892→            nextStep={nextStep}
   893→            dest={dest ? dest.position : null}
   894→            pickMode={pickMode}
   895→            onPick={pick}
   896→            userPin={userPin}
   897→            incidents={incidents}
   898→            locateTarget={locateTarget}
   899→            onUnavailable={() => setNativeFail(true)}
   900→          />
   901→        </div>
   902→
   903→        {/* Estado + leyenda + controles (debajo del lienzo) */}
   904→        <div className="flex items-center gap-2 flex-wrap">
   905→          {nav.loading && (
   906→            <span className="rounded-full bg-zinc-950/90 border border-sky-700 px-2.5 py-1 text-[10px] font-semibold text-sky-300 flex items-center gap-1.5">
   907→              <span className="inline-block h-2 w-2 rounded-full bg-sky-400 animate-pulse" />
   908→              Calculando ruta…
   909→            </span>
   910→          )}
   911→          {onRoute && !nav.loading && <LegendChip className="rounded-md bg-zinc-950/90 border border-zinc-800 px-2 py-1 flex items-center gap-2" route={route} />}
   912→          <Controls
   913→            className="ml-auto flex gap-1.5"
