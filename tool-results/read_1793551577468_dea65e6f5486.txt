   136→   136→function NavCamera({ position, bearing, target, zoom }: { position: LatLng | null; bearing: number; target: LatLng; zoom: number }) {
   137→   137→  const map = useMap();
   138→   138→  const init = useRef(false);
   139→   139→  useEffect(() => {
   140→   140→    if (!position) return;
   141→   141→    map.invalidateSize();
   142→   142→    const b = wrap360(bearing);
   143→   143→    if (!init.current) {
   144→   144→      init.current = true;
   145→   145→      map.setBearing(b);
   146→   146→      map.setView([target.lat, target.lon], zoom, { animate: false });
   147→   147→      return;
   148→   148→    }
   149→   149→    map.setBearing(b, { animate: true, duration: 0.4 });
   150→   150→    map.setView([target.lat, target.lon], zoom, { animate: true, duration: 0.6 });
   151→   151→  }, [position, map]); // target/zoom/derivados del mismo tick de posición
   152→   152→  return null;
   153→   153→}

   155→   155→/** Mantiente el tamaño interno de Leaflet sincronizado con el DOM:
   156→   156→ *  el minimapa cambia de alto al activarse la ruta (h-52 → h-80) y su
   157→   157→ *  contenedor interno es 135% más alto (por el tilt 3D). Sin esto, la
   158→   158→ *  cámara calcula con un tamaño viejo y la ruta/cámara se ven mal. */
   159→   159→function AutoInvalidate() {
   160→   160→  const map = useMap();
   161→   161→  useEffect(() => {
   162→   162→    const el = map.getContainer();
   163→   163→    const ro = new ResizeObserver(() => map.invalidateSize());
   164→   164→    ro.observe(el);
   165→   165→    return () => ro.disconnect();
   166→   166→  }, [map]);
   167→   167→  return null;
   168→   168→}

   170→   170→/** Ajusta la vista a la ruta completa cuando aún no hay posición.
   171→   171→ *  El contenedor interno es 135% más alto que la ventana visible (por el
   172→   172→ *  tilt 3D), así que compensa la franja superior recortada para que la
   173→   173→ *  ruta completa —del inicio al fin— quede dentro de la zona visible. */
   174→   174→function FitRoute({ coords }: { coords: LatLng[] | null }) {
   175→   175→  const map = useMap();
   176→   176→  useEffect(() => {
   177→   177→    if (!coords || coords.length < 2) return;
   178→   178→    map.invalidateSize();
   179→   179→    if (map.getBearing && map.getBearing() !== 0) map.setBearing(0);
   180→   180→    const tall = map.getContainer().clientHeight || 432;
   181→   181→    const hiddenTop = Math.round(tall - tall / 1.35); // franja bajo el banner que no se ve
   182→   182→    const bounds = L.latLngBounds(coords.map((c) => [c.lat, c.lon] as [number, number]));
   183→   183→    map.fitBounds(bounds, { paddingTopLeft: [24, hiddenTop + 8], paddingBottomRight: [24, 112] });
   184→   184→  }, [coords, map]);
   185→   185→  return null;
   186→   186→}

   188→   188→/** Desplaza un punto `distM` metros según un rumbo (para adelantar la
   189→   189→ *  cámara en modo libre y dejar el pin abajo, estilo navegación). */
   190→   190→function offsetAlong(p: LatLng, headingDeg: number, distM: number): LatLng {
   191→   191→  const r = (headingDeg * Math.PI) / 180;
   192→   192→  const mPerDegLat = 111_320;
   193→   193→  const mPerDegLon = 111_320 * Math.cos((p.lat * Math.PI) / 180);
   194→   194→  return {
   195→   195→    lat: p.lat + (Math.cos(r) * distM) / mPerDegLat,
   196→   196→    lon: p.lon + (Math.sin(r) * distM) / mPerDegLon,
   197→   197→  };
   198→   198→}

   419→   419→          <>
   420→   420→            {!onRoute && accuracyM !== null && accuracyM < 100 && (
   421→   421→              <Circle
   422→   422→                center={[position.lat, position.lon]}

   424→   424→                pathOptions={{ color: "#ffffff", weight: 1, fillColor: "#ffffff", fillOpacity: 0.08 }}

   429→   429→              position={[position.lat, position.lon]}

   439→   439→          <FitRoute coords={route!.coords} />

   444→   444→      </MapContainer>
   445→   445→      </div>
   446→   446→
   447→   447→      {/* ===== Banner de maniobra (modo ruta), estilo Waze ===== */}

   457→   457→            </div>
   458→   458→            {offRoute && <div className="text-[11px] text-amber-300 font-semibold mt-0.5">desviado, recalculando…</div>}

   460→   460→          {laterStep && laterStep.icon !== "end" && <TurnArrow icon={laterStep.icon} className="h-7 w-7 shrink-0 mt-1 text-zinc-500" />}
   461→   461→        </div>
   462→   462→      )}
   463→   463→
   464→   464→      {/* Leyenda compacta de la ruta (solo modo ruta) */}

   466→   466→        <div className="absolute left-2 top-[92px] z-[500] rounded-md bg-zinc-950/90 border border-zinc-800 px-2 py-1 flex items-center gap-2 pointer-events-none">

   474→   474→

   478→   478→        <div className="absolute bottom-[108px] inset-x-0 z-[500] flex justify-center pointer-events-none">

   490→   490→          title="Centrar en mi ubicación"

   502→   502→        >

   515→   515→      </div>
   516→   516→
   517→   517→      {/* ===== Barra inferior tipo Waze — SIN velocímetro: la velocidad
   518→   518→          vive en el velocímetro principal, no sobre el mapa ===== */}

   521→   521→          <div className="min-w-0 flex-1 text-center">
   522→   522→            <div className="font-mono text-xl font-bold text-zinc-50 leading-none tabular-nums">{etaMin > 0 ? arrivalClock : "—"}</div>
   523→   523→            <div className="text-[11px] text-zinc-400 leading-tight mt-0.5 truncate">

   606→   606→/** Vuela a un punto una sola vez (localización sin ride activo). */
   607→   607→function FollowOnce({ target }: { target: LatLng }) {
   608→   608→  const map = useMap();
   609→   609→  useEffect(() => {
   610→   610→    map.flyTo([target.lat, target.lon], Math.max(map.getZoom(), 16), { duration: 0.8 });
   611→   611→  }, [target, map]);
   612→   612→  return null;
   613→   613→}
