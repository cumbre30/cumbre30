   227→function NavCanvas(props: NavCanvasProps) {
   228→  const boxRef = useRef<HTMLDivElement>(null);
   229→  const mapRef = useRef<MlMap | null>(null);
   230→  const [ready, setReady] = useState(false);
   231→  const firstFixRef = useRef(false);
   232→
   233→  // refs vivas para handlers registrados una sola vez
   234→  const pickModeRef = useRef(props.pickMode);
   235→  const onPickRef = useRef(props.onPick);
   236→  useEffect(() => {
   237→    pickModeRef.current = props.pickMode;
   238→    onPickRef.current = props.onPick;
   239→  }, [props.pickMode, props.onPick]);
   240→
   241→  // ---- init ----
   242→  useEffect(() => {
   243→    if (!boxRef.current || mapRef.current) return;
   244→    const map = new maplibregl.Map({
   245→      container: boxRef.current,
   246→      style: navStyle(),
   247→      center: PARK_CENTER,
   248→      zoom: 13.5,
   249→      pitch: 0,
   250→      bearing: 0,
   251→      attributionControl: false,
   252→      dragRotate: false,
   253→      dragPan: false,
   254→      scrollZoom: false,
   255→      doubleClickZoom: false,
   256→      touchZoomRotate: false,
   257→      keyboard: false,
   258→      boxZoom: false,
   259→      maxPitch: 70,
   260→    });
   261→    mapRef.current = map;
   262→    map.on("load", () => {
   263→      map.addImage("c30-chev", makeChevronImage());
   264→
   265→      // asa de depuración/E2E (no interfiere en producción)
   266→      (window as unknown as Record<string, unknown>).__c30nav = map;
   267→
   268→      // --- fuentes ---
   269→      map.addSource("park", { type: "geojson", data: toFc([polyFeat(PARK_POLYGON.map((p) => [p.lon, p.lat] as [number, number]))]) });
   270→      map.addSource("risk", {
   271→        type: "geojson",
   272→        data: toFc(RISK_ZONES.map((z) => polyFeat(circlePolygon(z.position.lat, z.position.lon, z.radiusM), { id: z.id }))),
   273→      });
   274→      map.addSource("gates", {
   275→        type: "geojson",
   276→        data: toFc(GATES.map((g) => polyFeat(circlePolygon(g.position.lat, g.position.lon, 18), { id: g.id }))),
   277→      });
   278→      map.addSource("track", { type: "geojson", data: toFc([]) });
   279→      map.addSource("route-remaining", { type: "geojson", data: toFc([]) });
   280→      map.addSource("route-traveled", { type: "geojson", data: toFc([]) });
   281→
   282→      // --- capas ---
   283→      map.addLayer({ id: "park-fill", type: "fill", source: "park", paint: { "fill-color": "#10b981", "fill-opacity": 0.09 } });
   284→      map.addLayer({ id: "park-line", type: "line", source: "park", paint: { "line-color": "#34d399", "line-width": 1.4, "line-opacity": 0.8 } });
   285→      map.addLayer({ id: "risk-fill", type: "fill", source: "risk", paint: { "fill-color": "#f59e0b", "fill-opacity": 0.14 } });
   286→      map.addLayer({ id: "risk-line", type: "line", source: "risk", paint: { "line-color": "#f59e0b", "line-width": 1, "line-opacity": 0.7 } });
   287→      map.addLayer({ id: "gates-fill", type: "fill", source: "gates", paint: { "fill-color": "#0f766e", "fill-opacity": 0.9 } });
   288→      map.addLayer({
   289→        id: "track-line", type: "line", source: "track",
   290→        layout: { "line-cap": "round", "line-join": "round" },
   291→        paint: { "line-color": "#a3e635", "line-width": 3.5, "line-opacity": 0.9 },
   292→      });
   293→      map.addLayer({
   294→        id: "traveled-line", type: "line", source: "route-traveled",
   295→        filter: ["==", ["get", "ciclo"], true],
   296→        layout: { "line-cap": "round", "line-join": "round" },
   297→        paint: { "line-color": COLOR_CICLO, "line-width": 5, "line-opacity": 0.3 },
   298→      });
   299→      map.addLayer({
   300→        id: "traveled-line-via", type: "line", source: "route-traveled",
   301→        filter: ["==", ["get", "ciclo"], false],
   302→        layout: { "line-cap": "round", "line-join": "round" },
   303→        paint: { "line-color": COLOR_VIA, "line-width": 4, "line-opacity": 0.28 },
   304→      });
   305→      map.addLayer({
   306→        id: "route-casing", type: "line", source: "route-remaining",
   307→        filter: ["==", ["get", "kind"], "casing"],
   308→        layout: { "line-cap": "round", "line-join": "round" },
   309→        paint: { "line-color": COLOR_CASING, "line-width": ["interpolate", ["linear"], ["zoom"], 15, 9, 18, 14], "line-opacity": 0.9 },
   310→      });
   311→      map.addLayer({
   312→        id: "route-ciclo", type: "line", source: "route-remaining",
   313→        filter: ["==", ["get", "ciclo"], true],
   314→        layout: { "line-cap": "round", "line-join": "round" },
   315→        paint: { "line-color": COLOR_CICLO, "line-width": ["interpolate", ["linear"], ["zoom"], 15, 6.5, 18, 10], "line-opacity": 0.98 },
   316→      });
   317→      map.addLayer({
   318→        id: "route-via", type: "line", source: "route-remaining",
   319→        filter: ["all", ["==", ["get", "ciclo"], false], ["==", ["get", "kind"], "seg"]],
   320→        layout: { "line-cap": "round", "line-join": "round" },
   321→        paint: { "line-color": COLOR_VIA, "line-width": ["interpolate", ["linear"], ["zoom"], 15, 5, 18, 8], "line-opacity": 0.98 },
   322→      });
   323→      map.addLayer({
   324→        id: "route-arrows", type: "symbol", source: "route-remaining",
   325→        filter: ["==", ["get", "kind"], "seg"],
   326→        layout: {
   327→          "symbol-placement": "line",
   328→          "symbol-spacing": 130,
   329→          "icon-image": "c30-chev",
   330→          "icon-size": ["interpolate", ["linear"], ["zoom"], 15, 0.35, 18, 0.6],
   331→          "icon-rotation-alignment": "map",
   332→          "icon-allow-overlap": true,
   333→          "icon-ignore-placement": true,
   334→        },
   335→        paint: { "icon-opacity": 0.95 },
   336→      });
   337→
   338→      setReady(true);
   339→    });
   340→
   341→    map.on("click", (e) => {
   342→      if (pickModeRef.current) onPickRef.current({ lat: e.lngLat.lat, lon: e.lngLat.lng });
   343→    });
   344→
   345→    const ro = new ResizeObserver(() => map.resize());
   346→    ro.observe(boxRef.current);
   347→
   348→    return () => {
   349→      ro.disconnect();
   350→      map.remove();
   351→      mapRef.current = null;
   352→    };
   353→  }, []);
   354→
   355→  // ---- marcadores DOM ----
   356→  const puckRef = useRef<maplibregl.Marker | null>(null);
   357→  const arrowRef = useRef<HTMLElement | null>(null);
   358→  const puckPinRef = useRef<string | undefined>(undefined);
   359→  const flagRef = useRef<maplibregl.Marker | null>(null);
   360→  const badgeRef = useRef<maplibregl.Marker | null>(null);
   361→  const badgeIconRef = useRef<string | null>(null);
   362→  const incidentMarkersRef = useRef<maplibregl.Marker[]>([]);
   363→
   364→  // pin del usuario (recrear solo si cambia el pin elegido)
   365→  useEffect(() => {
   366→    const map = mapRef.current;
   367→    if (!map || !ready || !props.position) return;
   368→    if (!puckRef.current) {
   369→      const { el, arrow } = buildPuck(props.userPin);
   370→      puckPinRef.current = props.userPin;
   371→      arrowRef.current = arrow;
   372→      puckRef.current = new maplibregl.Marker({ element: el, rotationAlignment: "map" })
   373→        .setLngLat([props.position.lon, props.position.lat])
   374→        .addTo(map);
   375→    } else if (props.userPin !== puckPinRef.current) {
   376→      const { el, arrow } = buildPuck(props.userPin);
   377→      puckPinRef.current = props.userPin;
   378→      arrowRef.current = arrow;
   379→      puckRef.current.remove();
   380→      puckRef.current = new maplibregl.Marker({ element: el, rotationAlignment: "map" })
   381→        .setLngLat([props.position.lon, props.position.lat])
   382→        .addTo(map);
   383→    }
   384→  }, [ready, props.position, props.userPin]);
   385→
   386→  // posición viva del pin: sigue cada arreglo GPS
   387→  useEffect(() => {
   388→    if (puckRef.current && props.position) {
   389→      puckRef.current.setLngLat([props.position.lon, props.position.lat]);
   390→    }
   391→  }, [props.position]);
   392→
   393→  // bandera de destino + insignia del próximo giro
   394→  useEffect(() => {
   395→    const map = mapRef.current;
   396→    if (!map || !ready) return;
   397→    if (props.dest) {
   398→      if (!flagRef.current) {
   399→        flagRef.current = new maplibregl.Marker({ element: buildFlagEl() }).setLngLat([props.dest.lon, props.dest.lat]).addTo(map);
   400→      } else {
   401→        flagRef.current.setLngLat([props.dest.lon, props.dest.lat]);
   402→      }
   403→    } else if (flagRef.current) {
   404→      flagRef.current.remove();
   405→      flagRef.current = null;
   406→    }
   407→
   408→    const badge = props.nextStep && props.nextStep.icon !== "end" ? props.nextStep : null;
   409→    if (badge) {
   410→      if (!badgeRef.current) {
   411→        badgeRef.current = new maplibregl.Marker({ element: buildBadgeEl(badge.icon) }).setLngLat([badge.point.lon, badge.point.lat]).addTo(map);
   412→        badgeIconRef.current = badge.icon;
   413→      } else {
   414→        badgeRef.current.setLngLat([badge.point.lon, badge.point.lat]);
   415→        if (badgeIconRef.current !== badge.icon) {
   416→          // reemplazo del elemento: el DOM interno cambia de glifo
   417→          const oldEl = badgeRef.current.getElement();
   418→          const newEl = buildBadgeEl(badge.icon);
   419→          if (oldEl.parentNode) oldEl.parentNode.replaceChild(newEl, oldEl);
   420→          badgeIconRef.current = badge.icon;
   421→        }
   422→      }
   423→    } else if (badgeRef.current) {
   424→      badgeRef.current.remove();
   425→      badgeRef.current = null;
   426→      badgeIconRef.current = null;
   427→    }
   428→  }, [ready, props.dest, props.nextStep]);
   429→
   430→  // incidencias de la comunidad
   431→  useEffect(() => {
   432→    const map = mapRef.current;
   433→    if (!map || !ready) return;
   434→    for (const m of incidentMarkersRef.current) m.remove();
   435→    incidentMarkersRef.current = props.incidents.map((inc) =>
   436→      new maplibregl.Marker({ element: buildIncidentEl(inc.type) }).setLngLat([inc.lon, inc.lat]).addTo(map)
   437→    );
   438→  }, [ready, props.incidents]);
   439→
   440→  // ---- datos GeoJSON ----
   441→  useEffect(() => {
   442→    const map = mapRef.current;
   443→    if (!map || !ready) return;
   444→
   445→    // recorrido del ride
   446→    const trackSrc = map.getSource("track") as maplibregl.GeoJSONSource | undefined;
   447→    trackSrc?.setData(props.track.length > 1 ? toFc([lineFeat(props.track)]) : toFc([]));
   448→
   449→    if (!props.onRoute || !props.route) {
   450→      (map.getSource("route-remaining") as maplibregl.GeoJSONSource | undefined)?.setData(toFc([]));
   451→      (map.getSource("route-traveled") as maplibregl.GeoJSONSource | undefined)?.setData(toFc([]));
   452→      return;
   453→    }
   454→
   455→    const r = props.route;
   456→    // El tramo restante SIEMPRE parte de la ubicación ACTUAL: si el
   457→    // pin ya se movió del vértice más cercano, se antepone la
   458→    // posición viva como origen con un conector directo (así la ruta
   459→    // nace en tu punto, no en la ciclovía más próxima).
   460→    const feats = buildRouteFeatures(r, props.nearestIdx, props.position);
   461→
   462→    (map.getSource("route-remaining") as maplibregl.GeoJSONSource | undefined)?.setData(feats.remaining);
   463→    (map.getSource("route-traveled") as maplibregl.GeoJSONSource | undefined)?.setData(feats.traveled);
   464→  }, [ready, props.onRoute, props.route, props.nearestIdx, props.position, props.track]);
   465→
   466→  // ---- cámara: pin FIJO abajo y al centro, estilo Waze ----
   467→  useEffect(() => {
   468→    const map = mapRef.current;
   469→    if (!map || !ready) return;
   470→
   471→    // Sin posición: encuadra la ruta completa (vista previa plana)
   472→    if (!props.position) {
   473→      if (props.onRoute && props.route) {
   474→        const lons = props.route.coords.map((c) => c.lon);
   475→        const lats = props.route.coords.map((c) => c.lat);
   476→        map.easeTo({
   477→          center: [(Math.min(...lons) + Math.max(...lons)) / 2, (Math.min(...lats) + Math.max(...lats)) / 2],
   478→          zoom: 13.8,
   479→          pitch: 0,
   480→          bearing: 0,
   481→          duration: 700,
   482→        });
   483→      }
   484→      return;
   485→    }
   486→
   487→    // El CENTRO de la cámara es la posición del ciclista y el relleno
   488→    // superior corre el punto de ancla al ~30% del alto: el pin queda
   489→    // clavado abajo y centrado mientras el mundo fluye bajo él.
   490→    const h = boxRef.current?.clientHeight ?? 260;
   491→    const anchorBottom = Math.round(h * (props.onRoute ? 0.3 : 0.34));
   492→    const padding = { top: Math.max(0, Math.round(h) - anchorBottom * 2), bottom: 0, left: 0, right: 0 };
   493→
   494→    // rumbo: sensor de curso → dirección de la ruta → mantener actual
   495→    let b: number;
   496→    if (props.heading !== null) {
   497→      b = props.heading;
   498→    } else if (props.onRoute && props.route && props.nearestIdx !== null && props.nearestIdx < props.route.coords.length - 1) {
   499→      b = bearingOf(props.route.coords[props.nearestIdx], props.route.coords[props.nearestIdx + 1]);
   500→    } else {
   501→      b = map.getBearing();
   502→    }
   503→    const cur = map.getBearing();
   504→    // giro por el camino corto (evita vueltas de 340°)
   505→    while (b - cur > 180) b -= 360;
   506→    while (b - cur < -180) b += 360;
   507→
   508→    const camZoom = props.onRoute && props.remainingM < ARRIVAL_M ? ARRIVAL_ZOOM : NAV_ZOOM;
   509→
   510→    if (!firstFixRef.current) {
   511→      firstFixRef.current = true;
   512→      map.jumpTo({
   513→        center: [props.position.lon, props.position.lat],
   514→        bearing: b,
   515→        pitch: NAV_PITCH,
   516→        zoom: camZoom,
   517→        padding,
   518→      });
   519→      return;
   520→    }
   521→
   522→    map.easeTo({
   523→      center: [props.position.lon, props.position.lat],
   524→      bearing: b,
   525→      pitch: NAV_PITCH,
   526→      zoom: camZoom,
   527→      padding,
   528→      duration: 680,
   529→      easing: (t) => t,
   530→    });
   531→  }, [ready, props.position, props.heading, props.onRoute, props.remainingM, props.nearestIdx, props.route]);
   532→
   533→  // rotación del pin flecha respecto al mapa (queda alineado al rumbo)
   534→  useEffect(() => {
   535→    const map = mapRef.current;
   536→    if (!map || !ready) return;
   537→    const apply = () => {
   538→      const m = puckRef.current;
   539→      const arrow = arrowRef.current;
   540→      if (!m || !arrow) return;
   541→      const hdg = props.heading ?? 0;
   542→      arrow.style.transform = `rotate(${hdg - map.getBearing()}deg)`;
   543→    };
   544→    apply();
   545→    map.on("rotate", apply);
   546→    return () => {
   547→      map.off("rotate", apply);
   548→    };
   549→  }, [ready, props.heading, props.position]);
   550→
   551→  // vista a un punto (localización sin ride)
   552→  useEffect(() => {
   553→    const map = mapRef.current;
   554→    if (!map || !ready || !props.locateTarget) return;
   555→    map.flyTo({ center: [props.locateTarget.lon, props.locateTarget.lat], zoom: Math.max(map.getZoom(), 16), duration: 800 });
   556→  }, [ready, props.locateTarget]);
   557→
   558→  return (
   559→    // position absolute por estilo inline: el CSS de maplibre usa
   560→    // .maplibregl-map{position:relative} y pisa la utilidad `absolute`
   561→    <div ref={boxRef} style={{ position: "absolute", inset: 0 }} />
   562→  );
   563→}

   763→// ============================================================
   764→// Componente principal: lienzo + capa Waze (banner, barra, etc.)
   765→// ============================================================
   766→export default function MiniMap({ position, heading, track, accuracyM, userPin, nav, speedKmh = 0 }: Props) {
   767→  const incidents = useIncidentFeed();
   768→  const [pickMode, setPickMode] = useState(false);
   769→  const [locateTarget, setLocateTarget] = useState<LatLng | null>(null);
   770→  const [reportOpen, setReportOpen] = useState(false);
   771→  const [status, setStatus] = useState<string | null>(null);
   772→  const route = nav.route;
   773→  const dest = nav.dest;
   774→  // Mapa nativo (MapLibre Native vía plugin) o lienzo web (MapLibre GL JS)
   775→  // ⚠️ NATIVE_MAPS_ENABLED=false (v1.3.1): el WebView del dispositivo no
   776→  // renderiza ni el nativo ni MapLibre GL JS (WebGL caído) — se muestra un
   777→  // fondo de color sin mapa. Prioriza funcionamiento: en la app nativa
   778→  // corre Leaflet con tiles ráster (LeafletNav); el código nativo queda
   779→  // dormido detrás de este flag para depurarlo más adelante.
   780→  const NATIVE_MAPS_ENABLED = false;
   781→  const nativeDetected = useMemo(() => isNativeMapPlatform(), []);
   782→  const [nativeFail, setNativeFail] = useState(false);
   783→  const native = NATIVE_MAPS_ENABLED && nativeDetected && !nativeFail;
   784→
   785→  const webglOk = useMemo(() => {
   786→    try {
   787→      return webglSupported();
   788→    } catch {
   789→      return false;
   790→    }
   791→  }, []);
   792→
   793→  const pick = (p: LatLng) => {
   794→    nav.setDestination({ name: "Punto en el mapa", position: p });
   795→    setPickMode(false);
   796→  };
   797→
   798→  const locate = () => {
   799→    if (position) {
   800→      setLocateTarget(position);
   801→      return;
   802→    }
   803→    if ("geolocation" in navigator) {
   804→      navigator.geolocation.getCurrentPosition(
   805→        (pos) => setLocateTarget({ lat: pos.coords.latitude, lon: pos.coords.longitude }),
   806→        () => {},
   807→        { enableHighAccuracy: true, timeout: 8000 }
   808→      );
   809→    }
   810→  };
   811→
   812→  const onRoute = !!route && route.coords.length > 1;
   813→
   814→  const progress = useMemo(
   815→    () => (onRoute && position ? routeProgress(route!, position) : null),
   816→    [onRoute, route, position]
   817→  );
   818→
   819→  const nextStep = progress?.nextStep ?? route?.steps[0] ?? null;
   820→  const laterStep = onRoute && progress && route ? route.steps[progress.passedCount + 1] ?? null : null;
   821→  const offRoute = !!progress && progress.offRouteM > 140;
   822→  const remainingM = progress?.remainingM ?? route?.distanceM ?? 0;
   823→
   824→  const currentStreet = onRoute && progress ? streetAt(route!, progress.nearestIdx) : "";
   825→
   826→  const spd = speedKmh > 4 ? speedKmh : 15;
   827→  const etaMin = remainingM > 0 ? Math.max(1, Math.round((remainingM / 1000 / spd) * 60)) : 0;
   828→  const arrivalClock = useMemo(
   829→    () => new Date(Date.now() + etaMin * 60000).toLocaleTimeString("es-CL", { hour: "2-digit", minute: "2-digit", hourCycle: "h23" }),
   830→    [etaMin]
   831→  );
   832→
   833→  // ---- Reporte rápido de incidencias (botón amarillo) ----
   834→  const reportIncident = async (type: IncidentTypeId) => {
   835→    if (!position) {
   836→      setStatus("Aún sin GPS: no podemos fijar la ubicación del reporte.");
   837→      setReportOpen(false);
   838→      setTimeout(() => setStatus(null), 4000);
   839→      return;
   840→    }
   841→    setReportOpen(false);
   842→    const payload = { type, lat: position.lat, lon: position.lon };
   843→    try {
   844→      const r = await fetch("/api/incidents", {
   845→        method: "POST",
   846→        headers: { "Content-Type": "application/json" },
   847→        body: JSON.stringify(payload),
   848→      });
   849→      const d = (await r.json()) as { incident?: IncidentItem; error?: string };
   850→      if (!r.ok || !d.incident) throw new Error(d.error ?? "error");
   851→      addIncident(d.incident);
   852→      setStatus("Incidencia reportada ✓ visible para toda la comunidad");
   853→    } catch {
   854→      const local: IncidentItem = {
   855→        id: crypto.randomUUID(),
   856→        ...payload,
   857→        note: null,
   858→        createdAt: new Date().toISOString(),
   859→        username: "tú (local)",
   860→        authorPin: userPin ?? DEFAULT_PIN,
   861→        mine: true,
   862→        localOnly: true,
   863→      };
   864→      addIncident(local);
   865→      saveLocalIncidents([local, ...loadLocalIncidents()]);
   866→      setStatus("Sin conexión: reporte guardado en este dispositivo.");
   867→    }
   868→    setTimeout(() => setStatus(null), 5000);
   869→  };
   870→
   871→  // ---- ramas de render: NATIVO (MapLibre Native) o WEB (MapLibre GL JS) ----
   872→  if (native) {
   873→    return (
   874→      <div className="flex flex-col gap-1.5">
   875→        {/* Banner de maniobra ENCIMA del lienzo nativo */}
   876→        {onRoute && nextStep && (
   877→          <ManeuverBanner
   878→            className="rounded-xl border border-zinc-800 bg-zinc-950/95 px-3 py-2 flex items-center gap-3"
   879→            nextStep={nextStep}
   880→            distM={progress?.distToNextStepM ?? nextStep.distM}
   881→            destName={dest?.name ?? "Destino"}
   882→            offRoute={offRoute}
   883→            laterStep={laterStep}
   884→          />
   885→        )}
   886→
   887→        {/* SOLO el lienzo nativo dentro de la ventana: sin HTML encima */}
   888→        <div className={`relative rounded-xl overflow-hidden border border-zinc-800 shrink-0 ${onRoute ? "h-72" : "h-56"}`}>
   889→          <NativeMapCanvas
   890→            id="minimap"
   891→            mode="nav"
   892→            position={position}
   893→            heading={heading}
   894→            track={track}
   895→            accuracyM={accuracyM}
   896→            onRoute={onRoute}
   897→            route={onRoute ? route : null}
   898→            nearestIdx={progress ? progress.nearestIdx : null}
   899→            remainingM={remainingM}
   900→            nextStep={nextStep}
   901→            dest={dest ? dest.position : null}
   902→            pickMode={pickMode}
   903→            onPick={pick}
   904→            userPin={userPin}
   905→            incidents={incidents}
   906→            locateTarget={locateTarget}
   907→            onUnavailable={() => setNativeFail(true)}
   908→          />
   909→        </div>
   910→
   911→        {/* Estado + leyenda + controles (debajo del lienzo) */}
