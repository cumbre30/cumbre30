   124→/** Centra y sigue la posición en modo libre (norte arriba, zoom ciclócomputador). */
   125→function FollowFree({ position }: { position: LatLng | null }) {
   126→  const map = useMap();
   127→  const init = useRef(false);
   128→  useEffect(() => {
   129→    if (!position) return;
   130→    if (map.getBearing && map.getBearing() !== 0) map.setBearing(0);
   131→    if (!init.current) {
   132→      init.current = true;
   133→      map.setView([position.lat, position.lon], 17, { animate: false });
   134→      return;
   135→    }
   136→    map.setView([position.lat, position.lon], map.getZoom(), { animate: true, duration: 0.6 });
   137→  }, [position, map]);
   138→  return null;
   139→}

   141→/** Ajusta la vista a la ruta completa cuando aún no hay posición. */
   142→function FitRoute({ coords }: { coords: LatLng[] | null }) {
   143→  const map = useMap();
   144→  useEffect(() => {
   145→    if (!coords || coords.length < 2) return;
   146→    if (map.getBearing && map.getBearing() !== 0) map.setBearing(0);
   147→    const bounds = L.latLngBounds(coords.map((c) => [c.lat, c.lon] as [number, number]));
   148→    map.fitBounds(bounds, { paddingTopLeft: [16, 92], paddingBottomRight: [16, 84] });
   149→  }, [coords, map]);
   150→  return null;
   151→}

   153→/**
   154→ * Cámara de navegación: gira el mapa al rumbo (avance = arriba) y
   155→ * centra un punto de la ruta ~110 m adelante, dejando el pin abajo
   156→ * como en una app de navegación.
   157→ */
   158→function NavCamera({ position, bearing, target, zoom }: { position: LatLng | null; bearing: number; target: LatLng; zoom: number }) {
   159→  const map = useMap();
   160→  const init = useRef(false);
   161→  useEffect(() => {
   162→    if (!position) return;
   163→    const b = wrap360(bearing);
   164→    if (!init.current) {
   165→      init.current = true;
   166→      map.setBearing(b);
   167→      map.setView([target.lat, target.lon], zoom, { animate: false });
   168→      return;
   169→    }
   170→    map.setBearing(b, { animate: true, duration: 0.4 });
   171→    map.setView([target.lat, target.lon], zoom, { animate: true, duration: 0.6 });
   172→  }, [position, map]); // target/zoom/derivados del mismo tick de posición
   173→  return null;
   174→}

   176→/** Escala métrica (referencia tipo 200 m), elevada sobre la barra inferior. */
   177→function ScaleHandler() {
   178→  const map = useMap();
   179→  useEffect(() => {
   180→    const c = L.control.scale({ imperial: false, position: "bottomleft" });
   181→    c.addTo(map);
   182→    return () => {
   183→      c.remove();
   184→    };
   185→  }, [map]);
   186→  return null;
   187→}

   189→function ClickPick({ enabled, onPick }: { enabled: boolean; onPick: (p: LatLng) => void }) {
   190→  useMapEvents({
   191→    click(e) {
   192→      if (enabled) onPick({ lat: e.latlng.lat, lon: e.latlng.lng });
   193→    },
   194→  });
   195→  return null;
   196→}

   198→export default function MiniMap({ position, heading, track, accuracyM, userPin, nav, speedKmh = 0 }: Props) {
   199→  const incidents = useIncidentFeed();
   200→  const [pickMode, setPickMode] = useState(false);
   201→  const [locateTarget, setLocateTarget] = useState<LatLng | null>(null);
   202→  const [reportOpen, setReportOpen] = useState(false);
   203→  const [status, setStatus] = useState<string | null>(null);
   204→  const route = nav.route;
   205→  const dest = nav.dest;
   206→
   207→  const pick = (p: LatLng) => {
   208→    nav.setDestination({ name: "Punto en el mapa", position: p });
   209→    setPickMode(false);
   210→  };
   211→
   212→  const locate = () => {
   213→    if (position) {
   214→      setLocateTarget(position);
   215→      return;
   216→    }
   217→    if ("geolocation" in navigator) {
   218→      navigator.geolocation.getCurrentPosition(
   219→        (pos) => setLocateTarget({ lat: pos.coords.latitude, lon: pos.coords.longitude }),
   220→        () => {},
   221→        { enableHighAccuracy: true, timeout: 8000 }
   222→      );
   223→    }
   224→  };
   225→
   226→  const onRoute = !!route && route.coords.length > 1;
   227→
   228→  // Progreso vivo sobre la ruta (restante + desvío)
   229→  const progress = useMemo(
   230→    () => (onRoute && position ? routeProgress(route!, position) : null),
   231→    [onRoute, route, position]
   232→  );
   233→
   234→  const nextStep = progress?.nextStep ?? route?.steps[0] ?? null;
   235→  const laterStep = onRoute && progress && route ? route.steps[progress.passedCount + 1] ?? null : null;
   236→  const offRoute = !!progress && progress.offRouteM > 140;
   237→  const remainingM = progress?.remainingM ?? route?.distanceM ?? 0;
   238→
   239→  // Ruta COMPLETA (inicio → fin) partida por condición: recorrida (tenue) + restante (plena)
   240→  const traveledGroups = useMemo(
   241→    () => (onRoute && progress ? groupSegments(route!.coords, route!.flags, 0, progress.nearestIdx - 1) : []),
   242→    [onRoute, progress, route]
   243→  );
   244→  const remainingGroups = useMemo(
   245→    () => (onRoute ? groupSegments(route!.coords, route!.flags, progress ? progress.nearestIdx : 0, route!.coords.length - 2) : []),
   246→    [onRoute, progress, route]
   247→  );
   248→  const remainingLine = useMemo(
   249→    () => (onRoute ? route!.coords.slice(progress ? progress.nearestIdx : 0).map((c) => [c.lat, c.lon] as [number, number]) : []),
   250→    [onRoute, progress, route]
   251→  );
   252→
   253→  // Calle actual (píldora sobre el pin)
   254→  const currentStreet = onRoute && progress ? streetAt(route!, progress.nearestIdx) : "";
   255→
   256→  // ETA vivo: con la velocidad actual (toma 15 km/h como referencia base)
   257→  const spd = speedKmh > 4 ? speedKmh : 15;
   258→  const etaMin = remainingM > 0 ? Math.max(1, Math.round((remainingM / 1000) / spd * 60)) : 0;
   259→  const arrivalClock = useMemo(
   260→    () => new Date(Date.now() + etaMin * 60000).toLocaleTimeString("es-CL", { hour: "2-digit", minute: "2-digit", hourCycle: "h23" }),
   261→    [etaMin]
   262→  );
   263→
   264→  // ---- Reporte rápido de incidencias (botón amarillo) ----
   265→  const reportIncident = async (type: IncidentTypeId) => {
   266→    if (!position) {
   267→      setStatus("Aún sin GPS: no podemos fijar la ubicación del reporte.");
   268→      setReportOpen(false);
   269→      setTimeout(() => setStatus(null), 4000);
   270→      return;
   271→    }
   272→    setReportOpen(false);
   273→    const payload = { type, lat: position.lat, lon: position.lon };
   274→    try {
   275→      const r = await fetch("/api/incidents", {
   276→        method: "POST",
   277→        headers: { "Content-Type": "application/json" },
   278→        body: JSON.stringify(payload),
   279→      });
   280→      const d = (await r.json()) as { incident?: IncidentItem; error?: string };
   281→      if (!r.ok || !d.incident) throw new Error(d.error ?? "error");
   282→      addIncident(d.incident);
   283→      setStatus("Incidencia reportada ✓ visible para toda la comunidad");
   284→    } catch {
   285→      const local: IncidentItem = {
   286→        id: crypto.randomUUID(),
   287→        ...payload,
   288→        note: null,
   289→        createdAt: new Date().toISOString(),
   290→        username: "tú (local)",
   291→        authorPin: userPin ?? DEFAULT_PIN,
   292→        mine: true,
   293→        localOnly: true,
   294→      };
   295→      addIncident(local);
   296→      saveLocalIncidents([local, ...loadLocalIncidents()]);
   297→      setStatus("Sin conexión: reporte guardado en este dispositivo.");
   298→    }
   299→    setTimeout(() => setStatus(null), 5000);
   300→  };
   301→
   302→  // ----- Cámara tipo navegación -----
   303→  const navBearing = onRoute && heading !== null ? wrap360(-heading) : 0;
   304→  const camTarget: LatLng = useMemo(() => {
   305→    if (!onRoute || !position) return position ?? PARK_CENTER_OBJ;
   306→    if (offRoute || progress!.remainingM < ARRIVAL_M) return position;
   307→    return pointAhead(route!.coords, progress!.nearestIdx, LOOKAHEAD_M);
   308→  }, [onRoute, position, offRoute, progress, route]);
   309→  const camZoom = onRoute && progress && progress.remainingM < ARRIVAL_M ? ARRIVAL_ZOOM : NAV_ZOOM;
   310→
   311→  return (
   312→    <div className={`relative rounded-xl overflow-hidden border border-zinc-800 z-0 c30-minimap ${onRoute ? "h-72" : "h-52 sm:h-56"}`}>
   313→      <MapContainer
   314→        center={position ? [position.lat, position.lon] : PARK_CENTER}
   315→        zoom={position ? 17 : 13}
   316→        zoomControl={false}
   317→        attributionControl={false}
   318→        rotate
   319→        bearing={0}
   320→        rotateControl={false}
   321→        touchRotate={false}
   322→        dragging={false}
   323→        scrollWheelZoom={false}
   324→        doubleClickZoom={false}
   325→        touchZoom={false}
   326→        boxZoom={false}
   327→        keyboard={false}
   328→        className={`map-dark h-full w-full bg-zinc-950 ${pickMode ? "cursor-crosshair" : ""}`}
   329→      >
   330→        <TileLayer url="https://tile.openstreetmap.org/{z}/{x}/{y}.png" maxZoom={19} />
   331→
   332→        {!onRoute && (
   333→          <>
   334→            <Polygon
   335→              positions={PARK_POLYGON.map((p) => [p.lat, p.lon])}
   336→              pathOptions={{ color: "#34d399", weight: 1.5, fillColor: "#10b981", fillOpacity: 0.07 }}
   337→              interactive={false}
   338→            />
   339→            {RISK_ZONES.map((z) => (
   340→              <Circle
   341→                key={z.id}
   342→                center={[z.position.lat, z.position.lon]}
   343→                radius={z.radiusM}
   344→                pathOptions={{ color: "#f59e0b", weight: 1, fillColor: "#f59e0b", fillOpacity: 0.13 }}
   345→                interactive={false}
   346→              />
   347→            ))}
   348→            {GATES.map((g) => (
   349→              <CircleMarker
   350→                key={g.id}
   351→                center={[g.position.lat, g.position.lon]}
   352→                radius={5}
   353→                pathOptions={{ color: "#2dd4bf", weight: 1.5, fillColor: "#0f766e", fillOpacity: 0.9 }}
   354→                interactive={false}
   355→              />
   356→            ))}
   357→            {track.length > 1 && (
   358→              <Polyline positions={track.map((p) => [p.lat, p.lon])} pathOptions={{ color: "#a3e635", weight: 3.5, opacity: 0.9 }} interactive={false} />
   359→            )}
   360→          </>
   361→        )}
   362→
   363→        {/* Incidencias de la comunidad: visibles en ambos modos */}
   364→        {incidents.map((inc) => (
   365→          <Marker key={inc.id} position={[inc.lat, inc.lon]} icon={miniIncidentIcon(inc.type)} interactive={false} />
   366→        ))}
   367→
   368→        {onRoute && (
   369→          <>
   370→            {/* base oscura bajo la ruta activa para que destaque */}
   371→            {remainingLine.length > 1 && (
   372→              <Polyline positions={remainingLine} pathOptions={{ color: COLOR_CASING, weight: 10, opacity: 0.85 }} interactive={false} />
   373→            )}
   374→            {/* recorrido ya hecho: mismos colores, tenue (ruta completa a la vista) */}
   375→            {traveledGroups.map((g, i) => (
   376→              <Polyline
   377→                key={`t${i}`}
   378→                positions={g.coords.map((c) => [c.lat, c.lon] as [number, number])}
   379→                pathOptions={{ color: g.ciclovia ? COLOR_CICLO : COLOR_VIA, weight: g.ciclovia ? 5 : 4, opacity: 0.32 }}
   380→                interactive={false}
   381→              />
   382→            ))}
   383→            {/* tramo restante: violeta = ciclovía · blanco = vía compartida */}
   384→            {remainingGroups.map((g, i) => (
   385→              <Polyline
   386→                key={`r${i}`}
   387→                positions={g.coords.map((c) => [c.lat, c.lon] as [number, number])}
   388→                pathOptions={{ color: g.ciclovia ? COLOR_CICLO : COLOR_VIA, weight: g.ciclovia ? 7 : 5.5, opacity: 0.98, lineCap: "round", lineJoin: "round" }}
   389→                interactive={false}
   390→              />
   391→            ))}
   392→            {nextStep && nextStep.icon !== "end" && (
   393→              <Marker position={[nextStep.point.lat, nextStep.point.lon]} icon={stepBadgeIcon(nextStep.icon)} interactive={false} />
   394→            )}
   395→            {dest && <Marker position={[dest.position.lat, dest.position.lon]} icon={flagIcon()} interactive={false} />}
   396→          </>
   397→        )}
   398→
   399→        {/* Posición actual: pin elegido por el usuario o flecha por defecto */}
   400→        {position && (
   401→          <>
   402→            {!onRoute && accuracyM !== null && accuracyM < 100 && (
   403→              <Circle
   404→                center={[position.lat, position.lon]}
   405→                radius={accuracyM}
   406→                pathOptions={{ color: "#ffffff", weight: 1, fillColor: "#ffffff", fillOpacity: 0.08 }}
   407→                interactive={false}
   408→              />
   409→            )}
   410→            <Marker
   411→              position={[position.lat, position.lon]}
   412→              icon={userPinIcon(userPin ?? DEFAULT_PIN, heading, onRoute ? navBearing : 0)}
   413→              zIndexOffset={500}
   414→            />
   415→          </>
   416→        )}
   417→
   418→        <ScaleHandler />
   419→        {!onRoute ? (
   420→          <FollowFree position={position} />
   421→        ) : position ? (
   422→          <NavCamera position={position} bearing={navBearing} target={camTarget} zoom={camZoom} />
   423→        ) : (
   424→          <FitRoute coords={onRoute ? route!.coords : null} />
   425→        )}
   426→        {locateTarget && !position && <FollowOnce target={locateTarget} />}
   427→        <ClickPick enabled={pickMode} onPick={pick} />
   428→      </MapContainer>
   429→
   430→      {/* ===== Banner de maniobra (modo ruta), estilo Waze ===== */}
   431→      {onRoute && nextStep && (
   432→        <div className="absolute top-0 inset-x-0 z-[500] bg-zinc-950/95 border-b border-zinc-800 px-3 pt-1.5 pb-2 flex items-center gap-3 pointer-events-none">
   433→          <TurnArrow icon={nextStep.icon} className="h-14 w-14 shrink-0 text-zinc-50" />
   434→          <div className="min-w-0 flex-1">
   435→            <div className="font-mono text-[26px] font-bold text-zinc-50 leading-none tabular-nums">
   436→              {nextStep.icon === "end" ? "Llegaste" : fmtKmM(progress?.distToNextStepM ?? nextStep.distM)}
   437→            </div>
   438→            <div className="text-lg font-bold text-sky-400 leading-tight truncate mt-0.5">
   439→              {nextStep.icon === "end" ? dest?.name ?? "Destino" : streetOfStep(nextStep.text)}
   440→            </div>
   441→            {offRoute && <div className="text-[11px] text-amber-300 font-semibold mt-0.5">desviado, recalculando…</div>}
   442→          </div>
   443→          {laterStep && laterStep.icon !== "end" && <TurnArrow icon={laterStep.icon} className="h-7 w-7 shrink-0 mt-1 text-zinc-500" />}
   444→        </div>
   445→      )}
   446→
   447→      {/* Leyenda compacta de la ruta (solo modo ruta) */}
   448→      {onRoute && (
   449→        <div className="absolute left-2 top-[92px] z-[500] rounded-md bg-zinc-950/90 border border-zinc-800 px-2 py-1 flex items-center gap-2 pointer-events-none">
   450→          <span className="inline-block h-[3px] w-4 rounded-full shrink-0" style={{ background: COLOR_CICLO }} />
   451→          <span className="text-[10px] text-zinc-300">Ciclovía {!!route?.cicloviaPct ? `${route.cicloviaPct}%` : ""}</span>
   452→          <span className="text-[10px] text-zinc-600">·</span>
   453→          <span className="inline-block h-[3px] w-4 rounded-full shrink-0" style={{ background: COLOR_VIA }} />
   454→          <span className="text-[10px] text-zinc-300">Compartida</span>
   455→        </div>
   456→      )}
   457→
   458→      {/* Píldora con la calle actual sobre el pin (modo ruta; se oculta
   459→          si vas desviado o en llegada, cuando el pin vuelve al centro) */}
   460→      {onRoute && currentStreet && !offRoute && !(progress && progress.remainingM < ARRIVAL_M) && (
   461→        <div className="absolute bottom-[108px] inset-x-0 z-[500] flex justify-center pointer-events-none">
   462→          <span className="rounded-full bg-zinc-100 text-zinc-900 text-[11px] font-semibold px-3 py-1 shadow-lg max-w-[70%] truncate">
   463→            {currentStreet}
   464→          </span>
   465→        </div>
   466→      )}
   467→
   468→      {/* Controles flotantes */}
   469→      <div className="absolute top-2 right-2 z-[500] flex flex-col gap-1.5">
   470→        <button
   471→          onClick={locate}
   472→          aria-label="Localizarme"
   473→          title="Centrar en mi ubicación"
   474→          className="h-8 w-8 rounded-md bg-zinc-950/90 border border-zinc-700 text-zinc-100 flex items-center justify-center hover:border-emerald-500"
   475→        >
   476→          <Crosshair className="h-4 w-4" />
   477→        </button>
   478→        <button
   479→          onClick={() => setPickMode((v) => !v)}
   480→          aria-label="Elegir destino en el minimapa"
   481→          title="Toca el minimapa para fijar destino"
   482→          className={`h-8 w-8 rounded-md border flex items-center justify-center ${
   483→            pickMode ? "bg-sky-600 border-sky-300 text-white" : "bg-zinc-950/90 border-zinc-700 text-zinc-100 hover:border-sky-500"
   484→          }`}
   485→        >
   486→          <Flag className="h-4 w-4" />
   487→        </button>
   488→        {dest && (
   489→          <button
   490→            onClick={() => nav.setDestination(null)}
   491→            aria-label="Quitar destino"
   492→            title="Quitar la ruta"
   493→            className="h-8 w-8 rounded-md bg-zinc-950/90 border border-zinc-700 text-zinc-100 flex items-center justify-center hover:border-red-500"
   494→          >
   495→            <X className="h-4 w-4" />
   496→          </button>
   497→        )}
   498→      </div>
   499→
   500→      {/* ===== Barra inferior tipo Waze ===== */}
   501→      <div className="absolute bottom-0 inset-x-0 z-[500] bg-zinc-950/95 border-t border-zinc-800 px-2.5 py-2 flex items-center gap-2.5">
   502→        {/* velocímetro circular */}
   503→        <div
   504→          className="h-11 w-11 rounded-full bg-zinc-900 border-2 border-zinc-700 flex flex-col items-center justify-center shrink-0"
   505→          title="Velocidad actual"
   506→        >
   507→          <span className={`font-mono text-base font-bold leading-none tabular-nums ${speedKmh > 30 ? "text-red-400" : "text-zinc-50"}`}>
   508→            {Math.round(speedKmh) || "—"}
   509→          </span>
   510→          <span className="text-[8px] text-zinc-500 leading-none mt-0.5">km/h</span>
   511→        </div>
   512→
   513→        {onRoute ? (
   514→          <div className="min-w-0 flex-1 text-center">
   515→            <div className="font-mono text-xl font-bold text-zinc-50 leading-none tabular-nums">{etaMin > 0 ? arrivalClock : "—"}</div>
   516→            <div className="text-[11px] text-zinc-400 leading-tight mt-0.5 truncate">
   517→              {etaMin > 0 ? `${etaMin} min · ${fmtKmM(remainingM)}` : "Ruta lista"}
   518→            </div>
   519→          </div>
   520→        ) : (
   521→          <div className="min-w-0 flex-1 flex items-center gap-1.5 rounded-lg border border-zinc-800 bg-zinc-900 px-2 py-1.5">
   522→            <RouteIcon className="h-3.5 w-3.5 text-sky-400 shrink-0" />
   523→            <select
   524→              aria-label="Ir a"
   525→              value={dest ? NAV_DESTINATIONS.find((d) => d.position === dest.position)?.id ?? "mapa" : ""}
   526→              onChange={(e) => {
   527→                const d = NAV_DESTINATIONS.find((x) => x.id === e.target.value);
   528→                if (d) nav.setDestination({ name: d.name, position: d.position });
   529→                else nav.setDestination(null);
   530→              }}
   531→              className="bg-transparent text-[11px] text-zinc-200 outline-none w-full min-w-0"
   532→            >
   533→              <option value="" className="bg-zinc-900">Ir a…</option>
   534→              {NAV_DESTINATIONS.map((d) => (
   535→                <option key={d.id} value={d.id} className="bg-zinc-900">{d.name}</option>
   536→              ))}
   537→              {dest && <option value="mapa" className="bg-zinc-900">Punto en el mapa</option>}
   538→            </select>
   539→          </div>
   540→        )}
   541→
   542→        {/* botón amarillo de reporte de incidencias */}
   543→        <button
   544→          onClick={() => setReportOpen((v) => !v)}
   545→          aria-label="Reportar incidencia en mi posición"
   546→          title="Reportar incidencia en mi posición"
   547→          className="relative h-11 w-11 rounded-xl bg-amber-400 hover:bg-amber-300 active:scale-95 transition text-zinc-950 flex items-center justify-center shrink-0"
   548→        >
   549→          <TriangleAlert className="h-6 w-6" strokeWidth={2.4} />
   550→          <Plus className="absolute bottom-0 right-0.5 h-3.5 w-3.5" strokeWidth={3.5} />
   551→        </button>
   552→      </div>
   553→
   554→      {/* Hoja de reporte rápido */}
   555→      {reportOpen && (
   556→        <div className="absolute bottom-[60px] inset-x-2 z-[650] rounded-xl bg-zinc-950/95 border border-zinc-700 shadow-2xl p-3">
   557→          <div className="flex items-center justify-between mb-2">
   558→            <span className="text-xs font-bold text-zinc-100">Reportar incidencia {position ? "en tu posición" : "(sin GPS aún)"}</span>
   559→            <button onClick={() => setReportOpen(false)} aria-label="Cerrar reporte" className="text-zinc-400 hover:text-zinc-100">
   560→              <X className="h-4 w-4" />
   561→            </button>
   562→          </div>
   563→          <div className="grid grid-cols-2 gap-1.5">
   564→            {INCIDENT_TYPES.map((t) => (
   565→              <button
   566→                key={t.id}
   567→                onClick={() => reportIncident(t.id)}
   568→                className="flex items-center gap-1.5 rounded-lg border border-zinc-800 bg-zinc-900 px-2 py-1.5 text-left text-[11px] text-zinc-200 hover:border-zinc-500"
   569→              >
   570→                <span aria-hidden>{t.emoji}</span>
   571→                <span className="truncate">{t.label}</span>
   572→              </button>
   573→            ))}
   574→          </div>
   575→          <p className="text-[10px] text-zinc-500 mt-2">Se registra con fecha y hora, y aparece en el mapa compartido.</p>
   576→        </div>
   577→      )}
   578→
   579→      {/* Confirmación / avisos */}
   580→      {status && (
   581→        <div className="absolute bottom-[60px] inset-x-3 z-[700] rounded-lg bg-zinc-900/95 border border-zinc-700 px-3 py-2 text-[11px] text-zinc-100 pointer-events-none shadow-xl">
   582→          {status}
   583→        </div>
   584→      )}
   585→
   586→      {/* Aviso de modo elección de destino */}
   587→      {pickMode && (
   588→        <div className={`absolute inset-x-2 z-[700] rounded-md bg-sky-950/95 border border-sky-600 px-2.5 py-1.5 text-[11px] text-sky-100 flex items-center justify-between ${onRoute ? "top-[86px]" : "top-2"}`}>
   589→          <span>Toca el minimapa para fijar tu destino</span>
   590→          <button onClick={() => setPickMode(false)} aria-label="Cancelar elección" className="text-sky-300 hover:text-white">
   591→            <X className="h-3.5 w-3.5" />
   592→          </button>
   593→        </div>
   594→      )}
   595→    </div>
   596→  );
   597→}

   599→/** Vuela a un punto una sola vez (localización sin ride activo). */
   600→function FollowOnce({ target }: { target: LatLng }) {
   601→  const map = useMap();
   602→  useEffect(() => {
   603→    map.flyTo([target.lat, target.lon], Math.max(map.getZoom(), 16), { duration: 0.8 });
   604→  }, [target, map]);
   605→  return null;
   606→}
